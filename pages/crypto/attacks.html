Linear Congruential Generator (LCG)

Formula: state = (a * state + c) mod m. Fully predictable given a few outputs.

# LCG Parameter Recovery from consecutive outputs
# Given: X[n+1] = (a * X[n] + c) mod m
# If m is known (e.g., 2^32), recover a and c from 3 outputs:

def crack_lcg(outputs, m):
    """Recover LCG parameters from 3 consecutive outputs."""
    x0, x1, x2 = outputs[0], outputs[1], outputs[2]
    # a = (x2 - x1) * modinv(x1 - x0, m) mod m
    # c = (x1 - a * x0) mod m
    diff = x1 - x0
    diff2 = x2 - x1
    a = (diff2 * pow(diff, -1, m)) % m
    c = (x1 - a * x0) % m
    return a, c

# Unknown modulus recovery from 4+ outputs
def crack_lcg_unknown_modulus(outputs):
    """Recover m from consecutive LCG outputs using GCD method."""
    from math import gcd
    diffs = [outputs[i+1] - outputs[i] for i in range(len(outputs)-1)]
    # T[n] = diffs[n+1]*diffs[n-1] - diffs[n]^2 should be multiple of m
    zeroes = []
    for i in range(1, len(diffs)-1):
        zeroes.append(diffs[i+1]*diffs[i-1] - diffs[i]*diffs[i])
    m = abs(zeroes[0])
    for z in zeroes[1:]:
        m = gcd(m, abs(z))
    return m

# Meet-in-the-Middle on Double Encryption
# C = E(k2, E(k1, P))  =>  E(k1, P) = D(k2, C)
# Build table of E(k1, P) for all k1, then check D(k2, C) against table

from Crypto.Cipher import DES
from itertools import product

def mitm_double_des(plaintext, ciphertext, key_bits=24):
    """MITM attack on reduced-keyspace double DES."""
    # Phase 1: Forward encryption table
    forward_table = {}
    for k1_int in range(2**key_bits):
        k1 = k1_int.to_bytes(8, 'big')  # Padded DES key
        cipher = DES.new(k1, DES.MODE_ECB)
        intermediate = cipher.encrypt(plaintext)
        forward_table[intermediate] = k1

    # Phase 2: Backward decryption lookup
    for k2_int in range(2**key_bits):
        k2 = k2_int.to_bytes(8, 'big')
        cipher = DES.new(k2, DES.MODE_ECB)
        intermediate = cipher.decrypt(ciphertext)
        if intermediate in forward_table:
            return forward_table[intermediate], k2

    return None

    """Verify with replay protection."""
    mac = data[-32:]
    message = data[:-32]
    # Verify MAC
    if not hmac.compare_digest(mac, hmac.new(key, message, hashlib.sha256).digest()):
        return None

Discrete Log Attacks

Baby-Step Giant-Step (BSGS)

Solve gx = h (mod p) in O(√n) time and space.

import math

def baby_step_giant_step(g, h, p, order=None):
    """Solve g^x = h mod p. Returns x."""
    if order is None:
        order = p - 1
    m = math.isqrt(order) + 1

    # Baby step: compute g^j for j = 0..m-1
    table = {}
    power = 1
    for j in range(m):
        table[power] = j
        power = (power * g) % p

    # Giant step: g^(-m) mod p
    factor = pow(g, -m, p)
    gamma = h
    for i in range(m):
        if gamma in table:
            return i * m + table[gamma]
        gamma = (gamma * factor) % p
    return None  # No solution found

Pohlig-Hellman

When the group order n has small factors: n = p1e1 · p2e2 · ... , solve DLP in each small subgroup then combine with CRT.

from sympy.ntheory import factorint
from sympy.ntheory.residues import discrete_log

def pohlig_hellman(g, h, p, order):
    """Pohlig-Hellman algorithm for smooth-order groups."""
    factors = factorint(order)
    residues = []
    moduli = []

    for prime, exp in factors.items():
        # Project into subgroup of order prime^exp
        pe = prime ** exp
        gi = pow(g, order // pe, p)
        hi = pow(h, order // pe, p)
        # Solve DLP in small subgroup
        xi = discrete_log(p, hi, gi, pe)
        residues.append(xi)
        moduli.append(pe)

    # Combine with CRT
    from sympy.ntheory.modular import crt
    _, x = crt(moduli, residues)
    return x

Bleichenbacher's Attack (RSA PKCS#1 v1.5)

Adaptive chosen-ciphertext attack on RSA PKCS#1 v1.5 padding. Uses a padding oracle to decrypt RSA ciphertexts without the private key.

StepDescription
1. BlindingFind s0 such that c·s0e mod n has valid padding
2. SearchingFind successive si values narrowing the range of m
3. NarrowingCompute intervals where m must lie
4. ConvergenceSingle value remains → m recovered

Typically requires ~1 million oracle queries. Mitigated by OAEP padding (PKCS#1 v2.x).

# Bleichenbacher oracle simulation
def pkcs1_oracle(ciphertext, d, n, key_size_bytes):
    """Returns True if decrypted message has valid PKCS#1 v1.5 padding."""
    m = pow(ciphertext, d, n)
    plaintext = m.to_bytes(key_size_bytes, 'big')
    # Valid padding: 0x00 0x02 [random non-zero bytes] 0x00 [message]
    return plaintext[0] == 0x00 and plaintext[1] == 0x02

def bleichenbacher_step2(c, e, n, oracle, s_prev, B):
    """Find next multiplier s such that c * s^e mod n has valid padding."""
    s = s_prev + 1
    while True:
        candidate = (c * pow(s, e, n)) % n
        if oracle(candidate):
            return s
        s += 1

def custom_encrypt(key_bytes, plaintext_bytes):
    """4-round Feistel-like cipher with 4-byte key and 8-byte block."""

def baby_step_giant_step(g, h, p, order):
    """Solve g^x = h mod p where x < order."""
    m = isqrt(order) + 1
    # Baby steps
    table = {}
    pw = 1
    for j in range(m):
        table[pw] = j
        pw = pw * g % p
    # Giant step factor
    inv = pow(g, -m, p)
    gamma = h
    for i in range(m):
        if gamma in table:
            return i * m + table[gamma]
        gamma = gamma * inv % p
    return None

def pohlig_hellman(g, h, p, n):
    """
    Solve g^x = h mod p, given group order n.
    Requires n to have small prime factors.
    """
    factors = factorint(n)
    residues = []
    moduli = []

    for q, e in factors.items():
        qe = q ** e
        # Project to subgroup of order q^e
        exp = n // qe
        gi = pow(g, exp, p)
        hi = pow(h, exp, p)

        # Solve in subgroup (for simplicity, handle q^1 case)
        if e == 1:
            xi = baby_step_giant_step(gi, hi, p, q)
        else:
            # General case: solve digit by digit
            xi = 0
            gamma = hi
            for k in range(e):
                gk = pow(g, n // q, p)  # generator of order q
                hk = pow(gamma, n // (q**(k+1)), p)
                dk = baby_step_giant_step(gk, hk, p, q)
                xi += dk * (q ** k)
                gamma = gamma * pow(pow(g, dk * (q**k), p), -1, p) % p

        moduli.append(qe)
        print(f"  Subgroup q={q}, e={e}: x ≡ {xi} (mod {qe})")

    return x % n

    x = 0

        yi = pow(Mi, -1, mi)

    return x % M
