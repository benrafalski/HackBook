HMAC Construction

HMAC provides message authentication using a shared secret key with a hash function.

HMAC(K, m) = H((K' ⊕ opad) || H((K' ⊕ ipad) || m))

# K' = key padded to block size
# ipad = 0x36 repeated to block size
# opad = 0x5c repeated to block size
import hmac
import hashlib

key = b"secret_key"
message = b"authenticate this"

# Compute HMAC-SHA256
mac = hmac.new(key, message, hashlib.sha256).hexdigest()
print(f"HMAC-SHA256: {mac}")

# Secure comparison (constant-time)
expected = "known_mac_value"
if hmac.compare_digest(mac, expected):
    print("Valid!")

Length Extension Attacks (Merkle-Damgård)

Merkle-Damgård hashes (MD5, SHA-1, SHA-256) process messages in blocks and output the internal state directly. If you know H(secret || message) and the length of secret, you can compute H(secret || message || padding || extension) without knowing the secret.

# Using HashPump
hashpump -s "original_signature" \
         -d "original_data" \
         -k 16 \          # secret key length
         -a "appended_data"

# Output: new signature + new message (with glue padding)

Vulnerable pattern: MAC = H(secret || message)
Secure alternative: Use HMAC or SHA-3 (sponge construction immune to length extension).

Merkle Trees

Binary tree of hashes. Leaf nodes are hashes of data blocks; parent nodes are hashes of concatenated children. Used in Git, Bitcoin, certificate transparency.

import hashlib

def merkle_root(leaves):
    """Compute Merkle root from list of data blocks."""
    nodes = [hashlib.sha256(leaf).digest() for leaf in leaves]
    while len(nodes) > 1:
        if len(nodes) % 2 == 1:
            nodes.append(nodes[-1])  # duplicate last if odd
        nodes = [
            hashlib.sha256(nodes[i] + nodes[i+1]).digest()
            for i in range(0, len(nodes), 2)
        ]
    return nodes[0].hex()

blocks = [b"tx1", b"tx2", b"tx3", b"tx4"]
print(merkle_root(blocks))

Proof of Work

Find a nonce such that H(challenge || nonce) starts with N zero bits. Used in Bitcoin mining and CTF challenges.

import hashlib
import itertools

def proof_of_work(challenge, difficulty):
    """Find nonce where SHA-256(challenge + nonce) has `difficulty` leading zero bits."""
    prefix = '0' * (difficulty // 4)
    for nonce in itertools.count():
        attempt = f"{challenge}{nonce}".encode()
        h = hashlib.sha256(attempt).hexdigest()
        if h.startswith(prefix):
            return nonce, h
    
challenge = "CTF2024"
nonce, h = proof_of_work(challenge, 20)
print(f"Nonce: {nonce}, Hash: {h}")

Lab 5: HMAC Implementation from Scratch

Medium

Objective: Implement HMAC-SHA256 from scratch following RFC 2104 and verify against Python's hmac module.

import hashlib
import hmac as hmac_module

def hmac_sha256(key: bytes, message: bytes) -> str:
    """HMAC-SHA256 implementation per RFC 2104."""
    block_size = 64  # SHA-256 block size in bytes
    
    # Step 1: If key > block_size, hash it
    if len(key) > block_size:
        key = hashlib.sha256(key).digest()
    
    # Step 2: Pad key to block_size
    key = key.ljust(block_size, b'\x00')
    
    # Step 3: Create inner and outer padded keys
    ipad = bytes(k ^ 0x36 for k in key)
    opad = bytes(k ^ 0x5c for k in key)
    
    # Step 4: inner_hash = H(ipad || message)
    inner_hash = hashlib.sha256(ipad + message).digest()
    
    # Step 5: outer_hash = H(opad || inner_hash)
    outer_hash = hashlib.sha256(opad + inner_hash).hexdigest()
    
    return outer_hash

# Test vectors
key = b"secret_key_for_hmac"
message = b"The quick brown fox jumps over the lazy dog"

# Our implementation
custom = hmac_sha256(key, message)

# Python stdlib
reference = hmac_module.new(key, message, hashlib.sha256).hexdigest()

print(f"Custom HMAC:    {custom}")
print(f"Reference HMAC: {reference}")
print(f"Match: {custom == reference}")

# Test with key longer than block size
long_key = b"A" * 128
custom2 = hmac_sha256(long_key, message)
reference2 = hmac_module.new(long_key, message, hashlib.sha256).hexdigest()

Lab 6: Birthday Attack Simulation

Medium

Objective: Demonstrate the birthday paradox by finding collisions in truncated hash spaces and measure collision probability empirically.

import hashlib
import os
import math
from collections import defaultdict
import time

def truncated_hash(data, bits):
    """Return first `bits` bits of SHA-256 as hex."""
    full_hash = hashlib.sha256(data).hexdigest()
    hex_chars = bits // 4
    return full_hash[:hex_chars]

def find_collision(bits):
    """Find a collision in a truncated hash space."""
    seen = {}
    attempts = 0
    while True:
        attempts += 1
        data = os.urandom(16)
        h = truncated_hash(data, bits)
        if h in seen and seen[h] != data:
            return attempts, data, seen[h], h
        seen[h] = data

# Task 2: Empirical collision probability
def collision_probability(n_items, hash_space):
    """Monte Carlo: probability of collision with n items in hash_space."""
    trials = 1000
    collisions = 0
    for _ in range(trials):
        seen = set()
        for _ in range(n_items):
            h = truncated_hash(os.urandom(8), 16)
            if h in seen:
                collisions += 1
                break
            seen.add(h)
    return collisions / trials

Lab 7: Rainbow Table Basics

Medium

Objective: Build a miniature rainbow table for 4-digit PINs, understand reduction functions, and crack hashes using the table.

import hashlib

def hash_func(plaintext):
    """MD5 hash of plaintext."""
    return hashlib.md5(plaintext.encode()).hexdigest()

def reduce_func(hash_val, position):
    """Reduction function: maps hash back to a 4-digit PIN."""
    # Use different bytes based on chain position to avoid merging
    num = int(hash_val[position*2:(position*2)+4], 16)
    return f"{num % 10000:04d}"

def build_rainbow_table(chain_length=100):
    """Build rainbow table for 4-digit PINs (0000-9999)."""
    table = {}  # {endpoint: startpoint}
    
    for start_num in range(0, 10000, 10):  # sample every 10th
        start = f"{start_num:04d}"
        current = start
        
        # Build chain
        for pos in range(chain_length):
            h = hash_func(current)
            current = reduce_func(h, pos % 8)
        
        endpoint = current
        table[endpoint] = start
    
    print(f"Rainbow table built: {len(table)} chains, length {chain_length}")
    return table

def crack_with_rainbow(target_hash, table, chain_length=100):
    """Look up a hash in the rainbow table."""
    for start_pos in range(chain_length - 1, -1, -1):
        # Reproduce chain from position start_pos
        current = reduce_func(target_hash, start_pos % 8)
        
        for pos in range(start_pos + 1, chain_length):
            h = hash_func(current)
            current = reduce_func(h, pos % 8)
        
        # Check if endpoint is in table
        if current in table:
            # Reproduce chain from start to find plaintext
            candidate = table[current]
            for pos in range(chain_length):
                h = hash_func(candidate)
                if h == target_hash:
                    return candidate
                candidate = reduce_func(h, pos % 8)
    
    return None

# Build and test
table = build_rainbow_table()

# Crack some PINs
test_pins = ["1337", "4242", "0007", "9999"]
for pin in test_pins:
    target = hash_func(pin)
    result = crack_with_rainbow(target, table)
    status = "✓" if result == pin else "✗"
    print(f"  {status} PIN={pin}, hash={target[:16]}..., cracked={result}")

    """Generate password candidates from base words."""
    mutations = []
    
    leet_map = {'a': '@', 'e': '3', 'i': '1', 'o': '0', 's': '$'}

    suffixes = ['!', '#', '123', '!@#', '1!']

        # Original + capitalization variants
        mutations.extend([word, word.capitalize(), word.upper()])

        for m in sorted(set(mutations)):
            f.write(m + '\n')
    
    print(f"Generated {len(set(mutations))} candidates -> {output_file}")

def solve_pow_prefix(challenge, prefix_hex, algo='sha256', charset=None):
    """
    Find suffix s such that H(challenge + s) starts with prefix_hex.
    Common CTF PoW format.
    """
    if charset is None:
        charset = string.ascii_letters + string.digits
    
    hash_func = getattr(hashlib, algo)
    challenge_bytes = challenge.encode() if isinstance(challenge, str) else challenge
    
    for length in range(1, 12):
        for combo in itertools.product(charset, repeat=length):
            suffix = ''.join(combo).encode()
            h = hash_func(challenge_bytes + suffix).hexdigest()
            if h.startswith(prefix_hex):
                return suffix.decode(), h
    return None, None

def solve_pow_bits(challenge, zero_bits, algo='sha256'):
    """
    Find nonce n such that H(challenge + n) has `zero_bits` leading zero bits.
    """
    hash_func = getattr(hashlib, algo)
    challenge_bytes = challenge.encode() if isinstance(challenge, str) else challenge
    zero_hex = zero_bits // 4
    remaining = zero_bits % 4
    
    nonce = 0
    while True:
        attempt = challenge_bytes + str(nonce).encode()
        h = hash_func(attempt).hexdigest()
        
        if h[:zero_hex] == '0' * zero_hex:
            if remaining == 0 or int(h[zero_hex], 16) < (16 >> remaining):
                return nonce, h
        nonce += 1

import hashlib
import os

class HashChainOTP:
    """
    Lamport OTP: Server stores H^n(seed). 
    User authenticates with H^(n-1)(seed), H^(n-2)(seed), etc.
    Server verifies by hashing once and comparing to stored value.
    """
    
    def __init__(self, seed=None, chain_length=100):
        self.seed = seed or os.urandom(16)
        self.chain_length = chain_length
        self.chain = self._build_chain()
        # Server stores the tip (last hash)
        self.server_stored = self.chain[-1]
        # User has the full chain, uses in reverse
        self.current_index = chain_length - 2
    
    def _hash(self, data):
        if isinstance(data, str):
            data = data.encode()
        return hashlib.sha256(data).digest()
    
    def _build_chain(self):
        """Build hash chain: H^0(seed), H^1(seed), ..., H^n(seed)"""
        chain = [self.seed]
        for _ in range(self.chain_length):
            chain.append(self._hash(chain[-1]))
        return chain
    
    def get_next_otp(self):
        """User generates next OTP (traversing chain backwards)."""
        if self.current_index < 0:
            raise ValueError("Chain exhausted! Generate new seed.")
        otp = self.chain[self.current_index]
        self.current_index -= 1
        return otp
    
    def verify(self, otp):
        """Server verifies OTP by hashing once and comparing."""
        expected = self.server_stored
        computed = self._hash(otp)
        
        if computed == expected:
            # Update stored value for next authentication
            self.server_stored = otp
            return True
        return False

Challenge 10: Multi-Hash Crack

Hard

You have a password that was hashed with 3 different algorithms (MD5, SHA-1, SHA-256). All three hashes are provided. Use the constraints together to crack it efficiently.

"""
From a breached database, one user's password was stored with multiple hashes:

MD5:    e99a18c428cb38d5f260853678922e03
SHA-1:  6367c48dd193d56ea7b0baad25b19455e529f5ee  
SHA-256: 6ca13d52ca70c883e0f0bb101e425a89e8624de51db2d2392593af6a84118090

All are hashes of the SAME plaintext password.
The password is 6 characters, alphanumeric only.

Use all three as verification to avoid false positives.
"""
import hashlib
import itertools
import string
import time

target_md5 = "e99a18c428cb38d5f260853678922e03"
target_sha1 = "6367c48dd193d56ea7b0baad25b19455e529f5ee"
target_sha256 = "6ca13d52ca70c883e0f0bb101e425a89e8624de51db2d2392593af6a84118090"

def multi_hash_crack():
    charset = string.ascii_lowercase + string.digits
    start = time.time()
    attempts = 0
    
    # Optimization: check cheapest hash (MD5) first, then verify others
    for length in range(1, 7):
        for combo in itertools.product(charset, repeat=length):
            attempts += 1
            candidate = ''.join(combo)
            candidate_bytes = candidate.encode()
            
            # Fast check: MD5 first
            if hashlib.md5(candidate_bytes).hexdigest() != target_md5:
                continue
            
            # MD5 matched! Verify with SHA-1
            if hashlib.sha1(candidate_bytes).hexdigest() != target_sha1:
                print(f"  MD5 false positive: {candidate}")
                continue
            
            # Verify with SHA-256
            if hashlib.sha256(candidate_bytes).hexdigest() != target_sha256:
                print(f"  SHA-1 false positive: {candidate}")
                continue
            
            elapsed = time.time() - start
            print(f"Password found: '{candidate}'")
            print(f"Attempts: {attempts:,}")
            print(f"Time: {elapsed:.2f}s")
            return candidate
    
    return None

# Alternative: use hashcat on fastest hash, then verify
print("Strategy: Crack MD5 first (fastest), verify with others\n")

# Hashcat approach:
# hashcat -m 0 -a 3 e99a18c428cb38d5f260853678922e03 ?a?a?a?a?a?a

password = multi_hash_crack()
# Flag: CTF{password}

The password is "abc123". Strategy: crack the fastest hash (MD5) first with hashcat mask attack (?l?l?l?d?d?d), then verify the result against SHA-1 and SHA-256. Having multiple hashes eliminates false positives and confirms the correct plaintext.
