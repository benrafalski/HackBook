readelf

Inspect ELF structure — headers, sections, symbols, relocations.

# File header (arch, entry point, type)
$ readelf -h binary
  Entry point address:               0x401060
  Machine:                           Advanced Micro Devices X86-64

# Section headers (find .text, .rodata, .bss sizes)
$ readelf -S binary

# Symbol table (functions, globals)
$ readelf -s binary | grep FUNC
  45: 0000000000401156    85 FUNC    GLOBAL DEFAULT   14 check_password
  46: 00000000004011ab    42 FUNC    GLOBAL DEFAULT   14 decrypt_flag

# Dynamic symbols (imported functions)
$ readelf --dyn-syms binary | grep -i "crypt\|strcmp\|rand"

# Program headers (segments, LOAD addresses)
$ readelf -l binary

objdump

Disassemble sections with Intel or AT&T syntax.

# Disassemble .text with Intel syntax
$ objdump -d -M intel binary

# Disassemble specific function
$ objdump -d -M intel binary | sed -n '/<check_password>:/,/^$/p'

# Show all sections with content
$ objdump -s -j .rodata binary

# Disassemble with source interleaved (if compiled with -g)
$ objdump -d -S -M intel binary

# Show relocations
$ objdump -R binary

nm

List symbols — find functions, globals, undefined imports.

# All symbols sorted by address
$ nm binary

# Only defined functions (T=text/code)
$ nm binary | grep " T "

# Undefined symbols (imports)
$ nm -u binary

# Demangle C++ names
$ nm -C binary | grep -i "check\|verify\|validate"

# Symbol types: T=text, D=data, B=bss, U=undefined, W=weak

Cross-References (XREFs)

XREFs are the backbone of static analysis — they tell you where data/functions are used.

# In Ghidra's decompiler, click a function/variable, press Ctrl+Shift+F
# Shows all callers of a function or readers/writers of data

# Types of XREFs:
# - Call references: function A calls function B
# - Data references: code reads/writes a global variable
# - String references: code references a string in .rodata

Strategy: Find interesting strings → XREF to code → Trace logic backwards from there.

Patching Binaries in Ghidra

  1. Navigate to the instruction to patch
  2. Ctrl+Shift+G → Patch Instruction (or right-click → Patch)
  3. Modify instruction (e.g., JNE → JE, or NOP out bytes)
  4. File → Export Program → Binary (choose Original format)

Recognizing C Constructs in Assembly

if/else

; C: if (x == 5) { do_a(); } else { do_b(); }
    cmp     dword [rbp-4], 5
    jne     .else_branch        ; jump if NOT equal → else
    call    do_a
    jmp     .end_if
.else_branch:
    call    do_b
.end_if:

Ghidra decompiler output:

if (local_c == 5) {
    do_a();
} else {
    do_b();
}

for loop

; C: for (int i = 0; i < 10; i++) { buf[i] ^= key; }
    mov     dword [rbp-4], 0          ; i = 0
.loop_top:
    cmp     dword [rbp-4], 10         ; i < 10?
    jge     .loop_end
    ; body
    mov     eax, [rbp-4]
    cdqe
    movzx   eax, byte [rbp+rax-30]    ; buf[i]
    xor     eax, 0x42                 ; ^= key
    mov     edx, eax
    mov     eax, [rbp-4]
    cdqe
    mov     [rbp+rax-30], dl          ; store back
    ; increment
    add     dword [rbp-4], 1          ; i++
    jmp     .loop_top
.loop_end:

while loop

; C: while (*ptr != 0) { ptr++; count++; }
.while_top:
    mov     rax, [rbp-8]             ; load ptr
    movzx   eax, byte [rax]          ; *ptr
    test    al, al                    ; == 0?
    je      .while_end
    add     qword [rbp-8], 1          ; ptr++
    add     dword [rbp-4], 1          ; count++
    jmp     .while_top
.while_end:

switch/case (Jump Table)

; C: switch(choice) { case 0: ... case 1: ... case 2: ... }
    mov     eax, [rbp-4]             ; choice
    cmp     eax, 2                    ; bounds check
    ja      .default                  ; if > max case, goto default
    lea     rdx, [rip + .jump_table]
    movsxd  rax, dword [rdx+rax*4]   ; load offset from jump table
    add     rax, rdx                  ; compute target
    jmp     rax                       ; indirect jump

.jump_table:
    .long   .case0 - .jump_table
    .long   .case1 - .jump_table
    .long   .case2 - .jump_table

Tip: In Ghidra, switch tables show as computed jumps. Use "Create Jump Table" if auto-analysis misses it.

Function Calls & Structs

; Calling convention (System V AMD64): rdi, rsi, rdx, rcx, r8, r9, then stack
; C: result = validate(username, password, attempt_count);
    mov     edx, [rbp-4]             ; arg3 = attempt_count
    lea     rsi, [rbp-0x30]           ; arg2 = password (stack buffer)
    lea     rdi, [rbp-0x60]           ; arg1 = username (stack buffer)
    call    validate
    mov     [rbp-8], eax              ; store return value

; Struct access:
; C: player->health = 100;  (struct offset +0x8)
    mov     rax, [rbp-8]             ; load struct pointer
    mov     dword [rax+8], 100        ; write to offset 0x8

; Array access:
; C: arr[i] = value;  (int arr[], so stride = 4)
    mov     eax, [rbp-4]             ; i
    cdqe
    mov     dword [rbp+rax*4-0x40], 0x41  ; arr[i] = 'A'

String Recovery Techniques

  • Stack strings: Characters moved individually to stack — defeats simple strings
  • XOR-encoded strings: Loop XORs buffer with key, then uses result
  • Split strings: Concatenated at runtime from fragments
  • RC4/custom decrypt: Encrypted blob decrypted at runtime
; Stack string: "FLAG" built char by char
    mov     byte [rbp-0x10], 0x46    ; 'F'
    mov     byte [rbp-0x0f], 0x4c    ; 'L'
    mov     byte [rbp-0x0e], 0x41    ; 'A'
    mov     byte [rbp-0x0d], 0x47    ; 'G'
    mov     byte [rbp-0x0c], 0x00    ; null terminator
// Ghidra decompiler output for XOR-encoded string
void decrypt_string(char *buf, int len) {
    for (int i = 0; i < len; i++) {
        buf[i] = buf[i] ^ 0x37;
    }
}

Control Flow Graphs (CFG)

  • Basic block: Straight-line code with one entry, one exit
  • Edges: Conditional (true/false), unconditional jumps, fall-through
  • Dominators: Block A dominates B if all paths to B pass through A
  • Back edges: Edges to a dominator = loops

Tools: Ghidra graph view (Space), IDA graph (Space), r2 VV mode, Binary Ninja MLIL graph.

Data Flow Analysis

  • Def-use chains: Track where a value is defined and where it's used
  • Taint analysis: Track user input through transformations
  • Slice: Extract only instructions that affect a specific variable
  • Ghidra highlight: Click variable → all uses highlighted in decompiler

Cross-Referencing Strings to Code

The most common static analysis workflow:

  1. Open Strings window (Ghidra: Search → For Strings, IDA: Shift+F12)
  2. Find interesting string ("Access granted", "Wrong password", "flag{")
  3. Double-click → jumps to string in .rodata
  4. View XREFs to that string (Ghidra: right-click → References → Show References To)
  5. Navigate to referencing code → now you're in the validation function
  6. Analyze the logic that leads to printing that string

Angr Symbolic Execution

Angr explores all paths symbolically to find inputs that reach a target state.

import angr
import claripy

# Basic: find input that reaches "success" path, avoid "failure"
proj = angr.Project("./crackme", auto_load_libs=False)

# Create symbolic input (e.g., 20-byte password)
password = claripy.BVS("password", 20 * 8)  # 20 bytes = 160 bits

# Set up initial state with symbolic stdin
state = proj.factory.entry_state(stdin=password)

# Create simulation manager
simgr = proj.factory.simgr(state)

# Explore: find address that prints "Correct!", avoid "Wrong!"
simgr.explore(
    find=0x4011d0,    # address of success branch
    avoid=0x4011f0    # address of failure branch
)

# Extract solution
if simgr.found:
    solution = simgr.found[0]
    password_bytes = solution.solver.eval(password, cast_to=bytes)
    print(f"Password: {password_bytes}")

# Advanced: constrain to printable ASCII
for i in range(20):
    byte = password.get_byte(i)
    state.solver.add(byte >= 0x20)
    state.solver.add(byte <= 0x7e)
# Angr with function hooks (skip anti-debug, time checks)
import angr

proj = angr.Project("./binary", auto_load_libs=False)

# Hook a function to always return 0 (e.g., skip ptrace check)
@proj.hook(0x401050, length=5)
def skip_antidebug(state):
    state.regs.eax = 0  # pretend ptrace returned 0

# Hook libc functions
proj.hook_symbol("time", angr.SIM_PROCEDURES["libc"]["time"]())

state = proj.factory.entry_state()
simgr = proj.factory.simgr(state)
simgr.explore(find=lambda s: b"flag{" in s.posix.dumps(1))

if simgr.found:
    print(simgr.found[0].posix.dumps(1))  # stdout contains flag

Use Angr: create 16 symbolic bytes, constrain to printable, explore for the "flag{" output path. Angr's constraint solver (Z3 underneath) handles the complex transforms automatically. Key tip: use auto_load_libs=False and constrain input to printable ASCII to reduce the search space.
