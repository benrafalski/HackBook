Self-Debugging (Fork + Ptrace)

#include <sys/ptrace.h>
#include <sys/wait.h>
#include <unistd.h>

void self_debug() {
    pid_t child = fork();
    if (child == 0) {
        // Child traces parent - blocks external debuggers
        ptrace(PTRACE_ATTACH, getppid(), NULL, NULL);
        waitpid(getppid(), NULL, 0);
        ptrace(PTRACE_CONT, getppid(), NULL, NULL);
        // Keep tracing forever
        while (1) {
            waitpid(getppid(), NULL, 0);
            ptrace(PTRACE_CONT, getppid(), NULL, NULL);
        }
    }
    // Parent continues - now protected from external attach
    sleep(1);  // wait for child to attach
}

Opaque Predicates

; x*(x+1) is always even → x*(x+1) % 2 == 0 is always true

Junk Bytes

; Insert bytes that look like valid instructions to confuse disassemblers
    jmp short skip
    db 0xE8             ; 0xE8 = CALL opcode, disassembler tries to decode
                        ; the next 4 bytes as an address
skip:
    ; real code here
    mov eax, 1
    ret

; Another pattern - overlapping with jz
    jz label + 1
    db 0xE9             ; if not taken, this is junk
label:
    db 0x90             ; if taken, lands here (NOP)
    mov eax, 42

Overlapping Instructions

; Same bytes decode differently depending on entry point
; Entry at 0x00:  EB 01       jmp $+3
;                 E8 ...      (skipped byte + start of next)
; Entry at 0x02:  5B          pop ebx (different instruction!)

addr_0: jmp addr_2 + 1    ; jump into middle of next "instruction"
addr_2: db 0xE8            ; fake CALL opcode
addr_3: pop ebx            ; real instruction starts here
        ; disassembler following addr_0 shows CALL
        ; but execution hits POP EBX

Virtualization-Based Obfuscation (VM Protect)

// Custom VM interpreter - code is bytecode, not native
typedef struct {
    int regs[8];
    unsigned char *ip;
    int stack[256];
    int sp;
} VM;

enum { VM_PUSH=1, VM_POP, VM_ADD, VM_XOR, VM_CMP, VM_JZ, VM_HALT };

void vm_execute(VM *vm, unsigned char *bytecode, int len) {
    vm->ip = bytecode;
    while (vm->ip < bytecode + len) {
        switch (*vm->ip++) {
            case VM_PUSH:
                vm->stack[vm->sp++] = *vm->ip++;
                break;
            case VM_POP:
                vm->regs[*vm->ip++] = vm->stack[--vm->sp];
                break;
            case VM_ADD:
                vm->stack[vm->sp-2] += vm->stack[vm->sp-1];
                vm->sp--;
                break;
            case VM_XOR:
                vm->stack[vm->sp-2] ^= vm->stack[vm->sp-1];
                vm->sp--;
                break;
            case VM_CMP:
                vm->stack[vm->sp-2] = (vm->stack[vm->sp-2] == vm->stack[vm->sp-1]);
                vm->sp--;
                break;
            case VM_HALT:
                return;
        }
    }
}

Code Signing Checks

// Binary verifies its own integrity
#include <stdio.h>
#include <string.h>

unsigned int compute_crc32(void *data, size_t len) {
    unsigned int crc = 0xFFFFFFFF;
    unsigned char *p = (unsigned char*)data;
    for (size_t i = 0; i < len; i++) {
        crc ^= p[i];
        for (int j = 0; j < 8; j++)
            crc = (crc >> 1) ^ (0xEDB88320 & -(crc & 1));
    }
    return ~crc;
}

// Store expected CRC in a separate section or at end of binary
// Compare at runtime - if patched, CRC won't match
extern char __executable_start;
extern char __etext;

    if (crc != 0xDEADBEEF) {  // pre-computed expected value

    }
}

#define POP  0x02
#define ADDC 0x03   // add constant to TOS
#define SUBC 0x04   // subtract constant from TOS
#define XORC 0x05   // xor TOS with constant
#define CMPS 0x06   // compare top two stack values, push result
#define JMPF 0x07   // jump if TOS is 0 (false)
#define LOAD 0x08   // push input[idx]
#define PRNT 0x09   // print flag
#define HALT 0x0A

int vm_execute(unsigned char *code, int code_len, const char *input) {
    int stack[64], sp = 0, ip = 0;

    while (ip < code_len) {
        switch (code[ip++]) {
            case PUSH: stack[sp++] = code[ip++]; break;
            case POP:  sp--; break;
            case ADDC: stack[sp-1] += code[ip++]; break;
            case SUBC: stack[sp-1] -= code[ip++]; break;
            case XORC: stack[sp-1] ^= code[ip++]; break;

            case LOAD: stack[sp++] = (unsigned char)input[code[ip++]]; break;
            case PRNT: return 1;
            case HALT: return 0;
        }
    }
    return 0;
}
