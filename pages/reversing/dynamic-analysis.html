class HexDump(gdb.Command):
    def __init__(self):
        super().__init__("hexdump", gdb.COMMAND_USER)

    def invoke(self, arg, from_tty):
        addr = gdb.parse_and_eval(arg)
        inferior = gdb.selected_inferior()
        mem = inferior.read_memory(int(addr), 64)
        for i in range(0, 64, 16):
            hex_str = ' '.join(f'{b:02x}' for b in mem[i:i+16])
            print(f"{int(addr)+i:#x}: {hex_str}")

# Trace all calls to a function
class TraceFunc(gdb.Breakpoint):
    def __init__(self, func):
        super().__init__(func)
        self.silent = True
        self.count = 0

    def stop(self):
        self.count += 1
        rdi = gdb.parse_and_eval("$rdi")
        rsi = gdb.parse_and_eval("$rsi")
        print(f"[{self.count}] {self.location}(rdi={rdi:#x}, rsi={rsi:#x})")
        return False

Unicorn Engine (Emulation)

from unicorn import *
from unicorn.x86_const import *

# Emulate a function
def emulate_func(code, address=0x400000):
    mu = Uc(UC_ARCH_X86, UC_MODE_64)

    # Map memory
    mu.mem_map(address, 0x1000)       # code
    mu.mem_map(0x7ff000, 0x1000)      # stack

    # Write code
    mu.mem_write(address, code)

    # Set registers
    mu.reg_write(UC_X86_REG_RSP, 0x7ff800)
    mu.reg_write(UC_X86_REG_RDI, 0x41414141)

    # Hook for tracing
    def hook_code(uc, address, size, user_data):
        print(f"  {address:#x}: executing {size} bytes")

    mu.hook_add(UC_HOOK_CODE, hook_code)

    # Emulate
    mu.emu_start(address, address + len(code))

    # Read result
    rax = mu.reg_read(UC_X86_REG_RAX)
    print(f"Result RAX = {rax:#x}")
    return rax

void parse_header(const char *data, int len) {

    if (data[0] != 'M' || data[1] != 'Z') return;

def count_instructions(guess):

        capture_output=True, text=True
