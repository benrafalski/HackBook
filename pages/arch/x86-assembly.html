Data Movement

mov  rax, rbx          ; rax = rbx
mov  rax, [rbx]        ; rax = *(uint64_t*)rbx   (load from memory)
mov  [rbx], rax        ; *(uint64_t*)rbx = rax   (store to memory)
mov  rax, 0x41         ; rax = immediate value

lea  rax, [rbx+rcx*4]  ; rax = rbx + rcx*4  (address calculation, NO memory access)
lea  rax, [rip+0x200]  ; RIP-relative addressing (common in PIE binaries)

push rax               ; RSP -= 8; *(RSP) = rax
pop  rbx               ; rbx = *(RSP); RSP += 8

xchg rax, rbx          ; swap rax and rbx

; movzx / movsx — extend smaller values
movzx eax, BYTE [rbx]  ; zero-extend byte to 32-bit
movsx rax, DWORD [rbx]  ; sign-extend dword to 64-bit

lea vs mov: lea computes the address but does NOT dereference it. Compilers often use lea for arithmetic: lea rax, [rdi+rdi*2] computes rdi * 3.

Arithmetic

add  rax, rbx          ; rax += rbx
sub  rax, 10           ; rax -= 10
inc  rcx               ; rcx++ (does NOT set CF)
dec  rcx               ; rcx-- (does NOT set CF)
neg  rax               ; rax = -rax (two's complement)

; Multiplication
imul rax, rbx          ; rax = rax * rbx (signed, truncated to 64-bit)
imul rax, rbx, 5       ; rax = rbx * 5   (three-operand form)
mul  rbx               ; RDX:RAX = RAX * RBX (unsigned, 128-bit result)
imul rbx               ; RDX:RAX = RAX * RBX (signed, 128-bit result)

; Division — divides RDX:RAX by operand
; MUST zero/sign-extend RAX into RDX first!
xor  rdx, rdx          ; zero RDX for unsigned division
div  rbx               ; RAX = RDX:RAX / RBX, RDX = remainder (unsigned)

cqo                    ; sign-extend RAX into RDX for signed division
idiv rbx               ; RAX = quotient, RDX = remainder (signed)

Logic & Bitwise

and  rax, rbx          ; rax &= rbx
or   rax, rbx          ; rax |= rbx
xor  rax, rax          ; rax = 0  (common idiom, shorter than mov rax, 0)
not  rax               ; rax = ~rax (bitwise complement)

; Shifts
shl  rax, 3            ; rax <<= 3   (logical left shift, = multiply by 8)
shr  rax, 1            ; rax >>= 1   (logical right shift, = unsigned divide by 2)
sal  rax, 2            ; arithmetic left shift (same as shl)
sar  rax, 1            ; arithmetic right shift (preserves sign bit)

; Rotates
rol  rax, 4            ; rotate left 4 bits
ror  rax, 4            ; rotate right 4 bits

Comparison & Jumps

cmp  rax, rbx          ; sets flags based on (rax - rbx) without storing result
test rax, rax          ; sets flags based on (rax AND rax) — common zero check

Stack Operations

push rax               ; RSP -= 8; [RSP] = rax
pop  rbx               ; rbx = [RSP]; RSP += 8

call func              ; push RIP (return addr); jmp func
ret                    ; pop RIP (jump to saved return address)

; enter/leave — frame setup/teardown (leave is common, enter is slow/rare)
enter 32, 0            ; push rbp; mov rbp, rsp; sub rsp, 32
leave                  ; mov rsp, rbp; pop rbp

Function Prologue / Epilogue

; Prologue — sets up stack frame
my_function:
    push rbp             ; save caller's frame pointer
    mov  rbp, rsp        ; set frame pointer to current stack top
    sub  rsp, 0x20       ; allocate 32 bytes for local variables

; ... function body ...
; locals accessed via [rbp-8], [rbp-16], etc.
; args (if on stack) via [rbp+16], [rbp+24], etc.

; Epilogue — tears down stack frame
    leave                ; mov rsp, rbp; pop rbp
    ret                  ; return to caller

String Operations

; String ops use RSI (source), RDI (destination), RCX (count)
; Direction: DF=0 → forward (cld), DF=1 → backward (std)

cld                    ; clear direction flag (forward)

; rep prefix — repeat RCX times
rep movsb              ; memcpy: copy RCX bytes from [RSI] to [RDI]
rep stosb              ; memset: fill RCX bytes at [RDI] with AL
rep lodsb              ; load RCX bytes from [RSI] into AL (one at a time)

; repz / repnz — repeat while condition
repz cmpsb             ; strcmp: compare [RSI] vs [RDI], stop on mismatch or RCX=0
repnz scasb            ; strchr: scan [RDI] for AL, stop on match or RCX=0

; After rep cmpsb: ZF=1 means strings are equal (up to RCX bytes)
; After repnz scasb: ZF=1 means AL was found, RDI points past it

Addressing Modes

mov rax, [rbx]                ; Direct: *(rbx)
mov rax, [rbx + 8]            ; Base + displacement: *(rbx + 8)
mov rax, [rbx + rcx]          ; Base + index: *(rbx + rcx)
mov rax, [rbx + rcx*8]        ; Base + scaled index: *(rbx + rcx*8)
mov rax, [rbx + rcx*4 + 0x10] ; Full SIB: *(rbx + rcx*4 + 0x10)
mov rax, [rip + 0x2000]       ; RIP-relative (PIE/PIC code)
mov rax, [0x601040]            ; Absolute address (non-PIE only)

; Common patterns in compiled code:
; Array access:    arr[i]       → [rax + rcx*ELEMENT_SIZE]
; Struct field:    s->field     → [rax + FIELD_OFFSET]
; Stack local:     local_var    → [rbp - OFFSET]
; Stack argument:  arg7+        → [rbp + 16 + N*8]

NASM Program Structure

section .data
    msg db "Hello, world!", 10    ; string + newline
    msg_len equ $ - msg           ; length = current pos - start

section .bss
    buffer resb 256               ; reserve 256 uninitialized bytes

section .text
    global _start

_start:
    ; write(1, msg, msg_len)
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel msg]
    mov rdx, msg_len
    syscall

    ; exit(0)
    mov rax, 60
    xor rdi, rdi
    syscall

Recognizing C Constructs in Assembly

; if (x > 5) { body } else { alt }
cmp  DWORD [rbp-4], 5
jle  .else_branch        ; jump if x <= 5 (inverted condition)
; ... body ...
jmp  .end_if
.else_branch:
; ... alt ...
.end_if:

; for (int i = 0; i < n; i++) { body }
mov  DWORD [rbp-4], 0     ; i = 0
jmp  .loop_cond
.loop_body:
; ... body ...
add  DWORD [rbp-4], 1     ; i++
.loop_cond:
mov  eax, DWORD [rbp-4]
cmp  eax, DWORD [rbp-8]   ; i < n
jl   .loop_body

; while (cond) { body }
jmp  .while_cond
.while_body:
; ... body ...
.while_cond:
; ... evaluate cond ...
jne  .while_body

; switch — typically a jump table
; Compiler may use a table of addresses at [base + rax*8]
; or a chain of cmp/je for small switches

; arr[i] = value
mov  eax, DWORD [rbp-4]         ; eax = i
cdqe                              ; sign-extend to rax
mov  DWORD [rbp-48+rax*4], 42   ; arr[i] = 42

; What does this function do?
; Prototype: int mystery(int *arr, int n)
; arr → RDI, n → ESI

mystery:

    test esi, esi
    jle  .L_zero

    int max = 0;  // note: this is a bug if all elements are negative!

            max = arr[i];
    }
    return max;

Lab 7: Inline Assembly in C (GCC Extended Asm)

Medium

Objective: Use GCC inline assembly to read CPU registers (CR0, CPUID) and perform operations that C can't express directly.

// inline_asm.c — gcc -O0 -o inline_asm inline_asm.c
#include <stdio.h>
#include <stdint.h>

// Basic: read the timestamp counter
uint64_t rdtsc_inline() {
    uint32_t lo, hi;
    __asm__ __volatile__ (
        "rdtsc"
        : "=a"(lo), "=d"(hi)  // outputs: EAX → lo, EDX → hi
        :                       // no inputs
        :                       // no clobbers
    );
    return ((uint64_t)hi << 32) | lo;
}

// CPUID: get processor brand string
void get_cpu_vendor(char vendor[13]) {
    uint32_t eax, ebx, ecx, edx;
    __asm__ __volatile__ (
        "cpuid"
        : "=a"(eax), "=b"(ebx), "=c"(ecx), "=d"(edx)
        : "a"(0)    // input: EAX = 0 (get vendor)
    );
    *(uint32_t *)&vendor[0] = ebx;
    *(uint32_t *)&vendor[4] = edx;
    *(uint32_t *)&vendor[8] = ecx;
    vendor[12] = '\0';
}

// Atomic compare-and-swap
int cas(int *ptr, int expected, int desired) {
    int result;
    __asm__ __volatile__ (
        "lock cmpxchg %[desired], %[ptr]"
        : "=a"(result), [ptr] "+m"(*ptr)
        : "a"(expected), [desired] "r"(desired)
        : "cc"
    );
    return result == expected;
}

int main() {
    uint64_t t1 = rdtsc_inline();
    uint64_t t2 = rdtsc_inline();
    printf("RDTSC delta: %lu cycles\n", t2 - t1);

    char vendor[13];
    get_cpu_vendor(vendor);
    printf("CPU Vendor: %s\n", vendor);

    int val = 42;
    int ok = cas(&val, 42, 99);
    printf("CAS(42→99): %s, val=%d\n", ok ? "success" : "fail", val);
    return 0;
}

  2. Add a function that reads the FLAGS register using pushfq; pop

// Non-atomic increment (BROKEN under concurrency)
void inc_normal(volatile int *p) {
    __asm__ __volatile__ ("incl %0" : "+m"(*p));

void inc_atomic(volatile int *p) {

        else

    }
    return NULL;
}

    pthread_create(&t2, NULL, thread_func, &use_lock);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    printf("Expected: %d\n", 2 * ITERATIONS);
    printf("Actual:   %d\n", counter);
    printf("Lost:     %d\n", 2 * ITERATIONS - counter);
