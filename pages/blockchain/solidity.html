Solidity Fundamentals

How Solidity Works

Solidity is a statically-typed, contract-oriented language compiled to EVM bytecode. The compilation flow:

  1. .sol source → Solidity Compiler (solc)
  2. Compiler outputs ABI (JSON interface) + bytecode (deployment code)
  3. Bytecode is deployed to the blockchain via a transaction
  4. Deployed contract lives at an address and executes in the EVM
# Compile with solc
solc --abi --bin MyContract.sol -o build/

# Or use Hardhat
npx hardhat compile

Functions

// view: reads state, no modification
function getBalance() public view returns (uint256) {
    return balances[msg.sender];
}

// pure: no state access at all
function add(uint a, uint b) public pure returns (uint) {
    return a + b;
}

// payable: can receive ETH
function deposit() public payable {
    balances[msg.sender] += msg.value;
}

// Modifiers: reusable preconditions
modifier onlyOwner() {
    require(msg.sender == owner, "Not owner");
    _;  // placeholder for function body
}

function withdraw() public onlyOwner {
    payable(owner).transfer(address(this).balance);
}

// Multiple return values
function getInfo() public view returns (uint256, address) {
    return (balance, owner);
}

function process(string calldata name) external pure returns (uint) {
    return bytes(name).length;
}

// memory = mutable copy
function modify(uint[] memory data) internal pure returns (uint[] memory) {
    data[0] = 999;
    return data;
}

// storage = reference to state variable
function update() internal {
    User storage user = users[msg.sender];  // modifies state directly
    user.balance += 100;
}

Events and Logging

event Transfer(address indexed from, address indexed to, uint256 value);
// indexed params = searchable in logs (max 3 per event)

function transfer(address to, uint256 amount) public {
    balances[msg.sender] -= amount;
    balances[to] += amount;
    emit Transfer(msg.sender, to, amount);
}

Events are stored in transaction logs (not contract storage) — cheap to emit, not accessible from contracts. Used by off-chain apps (ethers.js, TheGraph) to track activity.

Error Handling

// require: validate inputs/conditions, refunds remaining gas
require(msg.value >= price, "Insufficient payment");

// revert: same as require but for complex conditions
if (balance < amount) {
    revert InsufficientBalance(balance, amount);  // custom error (gas efficient)
}

// assert: check invariants, should never fail in correct code
// consumes all remaining gas on failure (use sparingly)
assert(totalSupply == sumOfAllBalances);

// Custom errors (Solidity 0.8.4+) — cheaper than string messages
error InsufficientBalance(uint256 available, uint256 required);
error Unauthorized();

Inheritance and Interfaces

// Interface: defines function signatures, no implementation
interface IERC20 {
    function transfer(address to, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
}

// Abstract contract: can have partial implementation
abstract contract Ownable {
    address public owner;
    modifier onlyOwner() { require(msg.sender == owner); _; }
    function renounceOwnership() public virtual onlyOwner {
        owner = address(0);
    }
}

// Inheritance: is keyword, multiple inheritance (C3 linearization)
contract MyToken is IERC20, Ownable {
    // must implement all interface functions
    function transfer(address to, uint256 amount) external override returns (bool) {
        // ...
    }
}

// super keyword calls parent implementation
contract Child is Parent {
    function foo() public override {
        super.foo();  // calls Parent.foo()
    }
}

Step 1: Implement ERC-20

Create contracts/MyToken.sol:

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract MyToken {
    string public name;
    string public symbol;
    uint8 public decimals = 18;
    uint256 public totalSupply;

    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);

    constructor(string memory _name, string memory _symbol, uint256 _initialSupply) {
        name = _name;
        symbol = _symbol;
        totalSupply = _initialSupply * 10 ** decimals;
        balanceOf[msg.sender] = totalSupply;
        emit Transfer(address(0), msg.sender, totalSupply);
    }

    function transfer(address to, uint256 amount) public returns (bool) {
        require(to != address(0), "Transfer to zero address");
        require(balanceOf[msg.sender] >= amount, "Insufficient balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        emit Transfer(msg.sender, to, amount);
        return true;
    }

    function approve(address spender, uint256 amount) public returns (bool) {
        allowance[msg.sender][spender] = amount;
        emit Approval(msg.sender, spender, amount);
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) public returns (bool) {
        require(allowance[from][msg.sender] >= amount, "Allowance exceeded");
        require(balanceOf[from] >= amount, "Insufficient balance");
        allowance[from][msg.sender] -= amount;
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        emit Transfer(from, to, amount);
        return true;
    }
}

Step 2: Proxy Contract
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract SimpleProxy {
    // EIP-1967 implementation slot
    bytes32 private constant IMPLEMENTATION_SLOT =
        bytes32(uint256(keccak256("eip1967.proxy.implementation")) - 1);
    bytes32 private constant ADMIN_SLOT =
        bytes32(uint256(keccak256("eip1967.proxy.admin")) - 1);

    constructor(address _implementation) {
        _setImplementation(_implementation);
        _setAdmin(msg.sender);
    }

    function upgradeTo(address _newImpl) external {
        require(msg.sender == _getAdmin(), "Not admin");
        _setImplementation(_newImpl);
    }

    function _setImplementation(address _impl) private {
        assembly { sstore(IMPLEMENTATION_SLOT, _impl) }
    }

    function _getImplementation() private view returns (address impl) {
        assembly { impl := sload(IMPLEMENTATION_SLOT) }
    }

    function _setAdmin(address _admin) private {
        assembly { sstore(ADMIN_SLOT, _admin) }
    }

    function _getAdmin() private view returns (address admin) {
        assembly { admin := sload(ADMIN_SLOT) }
    }

    fallback() external payable {
        address impl = _getImplementation();
        assembly {
            calldatacopy(0, 0, calldatasize())
            let result := delegatecall(gas(), impl, 0, calldatasize(), 0, 0)
            returndatacopy(0, 0, returndatasize())
            switch result
            case 0 { revert(0, returndatasize()) }
            default { return(0, returndatasize()) }
        }
    }

    receive() external payable {}
}
Step 3: Upgraded Implementation (V2)

Step 4: Test the Upgrade
const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("Proxy Upgrade", function () {
  it("upgrades implementation and retains state", async () => {
    const BoxV1 = await ethers.getContractFactory("BoxV1");
    const boxV1 = await BoxV1.deploy();

    const Proxy = await ethers.getContractFactory("SimpleProxy");
    const proxy = await Proxy.deploy(await boxV1.getAddress());

    // Interact via V1 ABI through proxy
    const proxiedV1 = BoxV1.attach(await proxy.getAddress());
    await proxiedV1.initialize(42);
    expect(await proxiedV1.getValue()).to.equal(42);

    // Deploy V2 and upgrade
    const BoxV2 = await ethers.getContractFactory("BoxV2");
    const boxV2 = await BoxV2.deploy();
    await proxy.upgradeTo(await boxV2.getAddress());

    // State persists, new function available
    const proxiedV2 = BoxV2.attach(await proxy.getAddress());
    expect(await proxiedV2.getValue()).to.equal(42);
    await proxiedV2.increment();
    expect(await proxiedV2.getValue()).to.equal(43);
  });
});

Lab 4: ERC-721 Implementation

Objective: Implement a basic ERC-721 NFT contract from scratch with minting, metadata, and transfer functionality.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract SimpleNFT {
    string public name = "HackBookNFT";
    string public symbol = "HBNFT";
    uint256 private _nextTokenId;

    mapping(uint256 => address) private _owners;
    mapping(address => uint256) private _balances;
    mapping(uint256 => address) private _tokenApprovals;
    mapping(uint256 => string) private _tokenURIs;

    event Transfer(address indexed from, address indexed to, uint256 indexed tokenId);
    event Approval(address indexed owner, address indexed approved, uint256 indexed tokenId);

    function mint(address to, string memory uri) public returns (uint256) {
        uint256 tokenId = _nextTokenId++;
        _owners[tokenId] = to;
        _balances[to]++;
        _tokenURIs[tokenId] = uri;
        emit Transfer(address(0), to, tokenId);
        return tokenId;
    }

    function ownerOf(uint256 tokenId) public view returns (address) {
        address owner = _owners[tokenId];
        require(owner != address(0), "Token doesn't exist");
        return owner;
    }

    function balanceOf(address owner) public view returns (uint256) {
        return _balances[owner];
    }

    function tokenURI(uint256 tokenId) public view returns (string memory) {
        require(_owners[tokenId] != address(0), "Token doesn't exist");
        return _tokenURIs[tokenId];
    }

    function transferFrom(address from, address to, uint256 tokenId) public {
        require(_owners[tokenId] == from, "Not owner");
        require(
            msg.sender == from || msg.sender == _tokenApprovals[tokenId],
            "Not authorized"
        );
        _owners[tokenId] = to;
        _balances[from]--;
        _balances[to]++;
        delete _tokenApprovals[tokenId];
        emit Transfer(from, to, tokenId);
    }

    function approve(address to, uint256 tokenId) public {
        require(msg.sender == _owners[tokenId], "Not owner");
        _tokenApprovals[tokenId] = to;
        emit Approval(msg.sender, to, tokenId);
    }
}
// test/SimpleNFT.test.js
describe("SimpleNFT", function () {
  it("mints and transfers NFTs", async () => {
    const [owner, addr1] = await ethers.getSigners();
    const NFT = await ethers.getContractFactory("SimpleNFT");
    const nft = await NFT.deploy();

    await nft.mint(owner.address, "ipfs://QmHash1");
    expect(await nft.ownerOf(0)).to.equal(owner.address);
    expect(await nft.tokenURI(0)).to.equal("ipfs://QmHash1");

    await nft.transferFrom(owner.address, addr1.address, 0);
    expect(await nft.ownerOf(0)).to.equal(addr1.address);
  });
});

Lab 6: Custom Modifiers and Access Control

Easy

Objective: Implement a role-based access control system using custom modifiers, supporting owner, admin, and user roles.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract RoleBasedAccess {
    address public owner;
    mapping(address => bool) public admins;
    mapping(address => bool) public users;

    event RoleGranted(address indexed account, string role);
    event RoleRevoked(address indexed account, string role);

    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;
    }

    modifier onlyAdmin() {
        require(admins[msg.sender] || msg.sender == owner, "Not admin");
        _;
    }

    modifier onlyUser() {
        require(users[msg.sender] || admins[msg.sender] || msg.sender == owner, "Not user");
        _;
    }

    constructor() {
        owner = msg.sender;
    }

    function grantAdmin(address account) public onlyOwner {
        admins[account] = true;
        emit RoleGranted(account, "admin");
    }

    function revokeAdmin(address account) public onlyOwner {
        admins[account] = false;
        emit RoleRevoked(account, "admin");
    }

    function grantUser(address account) public onlyAdmin {
        users[account] = true;
        emit RoleGranted(account, "user");
    }

    // Only users can access this
    function protectedAction() public onlyUser returns (string memory) {
        return "Access granted!";
    }

    // Only owner can self-destruct
    function destroy() public onlyOwner {
        selfdestruct(payable(owner));
    }
}

Lab 7: Event Indexing and Off-Chain Monitoring

Easy

Objective: Emit events with indexed parameters and write a JavaScript listener that filters for specific events in real time.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract EventLogger {
    event ItemCreated(uint256 indexed id, address indexed creator, string name, uint256 price);
    event ItemSold(uint256 indexed id, address indexed buyer, uint256 price);
    event PriceChanged(uint256 indexed id, uint256 oldPrice, uint256 newPrice);

    struct Item {
        address creator;
        string name;
        uint256 price;
        bool sold;
    }

    mapping(uint256 => Item) public items;
    uint256 public nextId;

    function createItem(string calldata name, uint256 price) external {
        uint256 id = nextId++;
        items[id] = Item(msg.sender, name, price, false);
        emit ItemCreated(id, msg.sender, name, price);
    }

    function buyItem(uint256 id) external payable {
        Item storage item = items[id];
        require(!item.sold, "Already sold");
        require(msg.value >= item.price, "Underpaid");
        item.sold = true;
        payable(item.creator).transfer(msg.value);
        emit ItemSold(id, msg.sender, msg.value);
    }
}
// scripts/listen.js — listen for events
const { ethers } = require("hardhat");

async function main() {
  const contract = await ethers.getContractAt("EventLogger", "DEPLOYED_ADDR");

  // Filter: only ItemCreated events from a specific creator
  const filter = contract.filters.ItemCreated(null, "0xYOUR_ADDRESS");
  contract.on(filter, (id, creator, name, price, event) => {
    console.log(`New item #${id}: ${name} for ${ethers.formatEther(price)} ETH`);
  });

  // Query past events
  const events = await contract.queryFilter("ItemSold", -1000);
  events.forEach(e => console.log(`Sold: item #${e.args.id}`));
}

Lab 8: Multi-Sig Wallet

Hard

Objective: Build a multi-signature wallet that requires M-of-N owner approvals before executing transactions.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract MultiSigWallet {
    address[] public owners;
    uint256 public required;  // minimum approvals needed

    struct Transaction {
        address to;
        uint256 value;
        bytes data;
        bool executed;
        uint256 approvalCount;
    }

    Transaction[] public transactions;
    mapping(uint256 => mapping(address => bool)) public approved;
    mapping(address => bool) public isOwner;

    modifier onlyOwner() { require(isOwner[msg.sender], "Not owner"); _; }

    constructor(address[] memory _owners, uint256 _required) {
        require(_owners.length >= _required && _required > 0, "Invalid config");
        for (uint i = 0; i < _owners.length; i++) {
            require(!isOwner[_owners[i]], "Duplicate owner");
            isOwner[_owners[i]] = true;
            owners.push(_owners[i]);
        }
        required = _required;
    }

    function submit(address to, uint256 value, bytes calldata data) external onlyOwner {
        transactions.push(Transaction(to, value, data, false, 0));
    }

    function approve(uint256 txId) external onlyOwner {
        require(!approved[txId][msg.sender], "Already approved");
        approved[txId][msg.sender] = true;
        transactions[txId].approvalCount++;
    }

    function execute(uint256 txId) external onlyOwner {
        Transaction storage txn = transactions[txId];
        require(!txn.executed, "Already executed");
        require(txn.approvalCount >= required, "Not enough approvals");

}

Lab 9: Timelock Contract

Objective: Build a timelock contract that queues transactions and enforces a delay before execution — a critical governance safety mechanism.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract Timelock {
    uint256 public constant MIN_DELAY = 1 days;
    uint256 public constant MAX_DELAY = 30 days;
    address public admin;

    mapping(bytes32 => bool) public queued;

    event Queued(bytes32 indexed txId, address target, uint256 value,
                 bytes data, uint256 executeAfter);
    event Executed(bytes32 indexed txId);

    constructor() { admin = msg.sender; }

    function getTxId(address target, uint256 value, bytes calldata data,
                     uint256 executeAfter) public pure returns (bytes32) {
        return keccak256(abi.encode(target, value, data, executeAfter));
    }

    function queue(address target, uint256 value, bytes calldata data,
                   uint256 executeAfter) external {
        require(msg.sender == admin, "Not admin");
        require(executeAfter >= block.timestamp + MIN_DELAY, "Too soon");
        require(executeAfter <= block.timestamp + MAX_DELAY, "Too late");

        bytes32 txId = getTxId(target, value, data, executeAfter);
        require(!queued[txId], "Already queued");
        queued[txId] = true;
        emit Queued(txId, target, value, data, executeAfter);
    }

    function execute(address target, uint256 value, bytes calldata data,
                     uint256 executeAfter) external payable {
        require(msg.sender == admin, "Not admin");
        bytes32 txId = getTxId(target, value, data, executeAfter);
        require(queued[txId], "Not queued");
        require(block.timestamp >= executeAfter, "Too early");
        queued[txId] = false;
        (bool ok, ) = target.call{value: value}(data);
        require(ok, "Execution failed");
        emit Executed(txId);
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

// Library: reusable utility functions
library MathLib {
    function clamp(uint256 value, uint256 min, uint256 max)
        internal pure returns (uint256) {
        if (value < min) return min;
        if (value > max) return max;
        return value;
    }

    function percentage(uint256 amount, uint256 bps)
        internal pure returns (uint256) {
        return (amount * bps) / 10000;  // basis points
    }

// Interface
interface IVault {
    function deposit() external payable;
    function withdraw(uint256 amount) external;
    function balanceOf(address user) external view returns (uint256);
}

// Implementation
contract Vault is IVault {
    using MathLib for uint256;

    mapping(address => uint256) private balances;
    uint256 public constant MAX_DEPOSIT = 100 ether;
    uint256 public constant FEE_BPS = 50;  // 0.5% fee

    function deposit() external payable override {
        uint256 amount = msg.value.clamp(0, MAX_DEPOSIT);
        uint256 fee = amount.percentage(FEE_BPS);
        balances[msg.sender] += amount - fee;
    }

    function withdraw(uint256 amount) external override {
        require(balances[msg.sender] >= amount, "Insufficient");
        balances[msg.sender] -= amount;
        payable(msg.sender).transfer(amount);
    }

    function balanceOf(address user) external view override returns (uint256) {
        return balances[user];
    }
}

Challenge 3: Gas Optimization Challenge

Medium

The following contract wastes gas in multiple ways. Refactor it so that processAll() uses at least 40% less gas while producing identical results.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

        bool active;        // wastes a full slot

Solidity Fundamentals

How Solidity Works

Solidity is a compiled language for writing smart contracts on EVM (Ethereum Virtual Machine) blockchains.

You write .sol files → Compiler (solc) → Bytecode → Deployed on blockchain

The flow:

  1. You write Solidity code (.sol files)
  2. The compiler (solc) translates it to bytecode (machine code for the EVM)
  3. Hardhat uses solc under the hood when you run npx hardhat compile
  4. The bytecode gets deployed to a blockchain as a smart contract
  5. Users interact with the contract by sending transactions

Uses ethers.js/viem + Chai assertions. Supports console.log in Solidity.
evmVersion: "cancun" for latest opcodes (OpenZeppelin v5).
// test/MyToken.test.js
const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("MyToken", function () {
  let token, owner, addr1;

  beforeEach(async function () {
    [owner, addr1] = await ethers.getSigners();
    const MyToken = await ethers.getContractFactory("MyToken");
    token = await MyToken.deploy(1000);       // deploy with constructor arg
  });

  it("should transfer tokens", async function () {
    await token.transfer(addr1.address, 50);
    expect(await token.balanceOf(addr1.address)).to.equal(50);
  });

  it("should revert on insufficient balance", async function () {
    await expect(token.connect(addr1).transfer(owner.address, 1))
      .to.be.revertedWith("Insufficient balance");
  });
});
// scripts/deploy.js
const { ethers } = require("hardhat");

async function main() {
  const MyToken = await ethers.getContractFactory("MyToken");
  const token = await MyToken.deploy(1000);
  await token.waitForDeployment();
  console.log("Deployed to:", await token.getAddress());
}
main();

Foundry (Rust-based CLI)

Fastest compilation. Tests written in Solidity. Powerful fuzzing built-in.

curl -L https://foundry.paradigm.xyz | bash ; foundryup ; forge init my-project

src/*.sol          test/*.t.sol          script/*.s.sol          foundry.toml

forge build                              → compile
forge test                               → run tests
forge test --match-test "name"           → run specific test
anvil                                    → start local chain
forge script script/Deploy.s.sol         → deploy script

Cheatcodes: vm.prank(addr), vm.deal(addr, amt), vm.warp(time), vm.expectRevert()
// test/MyToken.t.sol
import "forge-std/Test.sol";
import "../src/MyToken.sol";

contract MyTokenTest is Test {
    MyToken token;
    address alice = makeAddr("alice");

    function setUp() public {
        token = new MyToken(1000);
    }

    function testTransfer() public {
        token.transfer(alice, 50);
        assertEq(token.balanceOf(alice), 50);
    }

    function testRevertInsufficientBalance() public {
        vm.prank(alice);                         // next call is from alice
        vm.expectRevert("Insufficient balance");
        token.transfer(address(this), 1);
    }

    function testFuzz_Transfer(uint256 amount) public {
        vm.assume(amount <= 1000);              // bound fuzz input
        token.transfer(alice, amount);
        assertEq(token.balanceOf(alice), amount);
    }
}
// script/Deploy.s.sol
import "forge-std/Script.sol";
import "../src/MyToken.sol";

contract DeployScript is Script {
    function run() external {
        vm.startBroadcast();                     // start signing txs
        new MyToken(1000);
        vm.stopBroadcast();
    }
}

Brownie (Python)

Python-based framework using web3.py. Good for Python-native devs. (Largely superseded by Ape.)

pip install eth-brownie ; brownie init

contracts/*.sol          tests/*.py          scripts/*.py          brownie-config.yaml

brownie compile                → compile
brownie test                   → run pytest tests
brownie test -k "name"         → run specific test
brownie console                → interactive Python REPL with deployed contracts
brownie run scripts/deploy.py  → deploy script

Uses web3.py + pytest. accounts[0] for test wallets.
# tests/test_my_token.py
from brownie import MyToken, accounts, reverts

def test_transfer():
    token = MyToken.deploy(1000, {"from": accounts[0]})
    token.transfer(accounts[1], 50, {"from": accounts[0]})
    assert token.balanceOf(accounts[1]) == 50

def test_revert_insufficient():
    token = MyToken.deploy(1000, {"from": accounts[0]})
    with reverts("Insufficient balance"):
        token.transfer(accounts[0], 1, {"from": accounts[1]})
# scripts/deploy.py
from brownie import MyToken, accounts

def main():
    token = MyToken.deploy(1000, {"from": accounts[0]})
    print(f"Deployed to: {token.address}")

Ape (Python)

Modern Python framework, spiritual successor to Brownie. Plugin-based, cleaner API.

pip install eth-ape ; ape init

contracts/*.sol          tests/*.py          scripts/*.py          ape-config.yaml

ape compile                    → compile
ape test                       → run pytest tests
ape test -k "name"             → run specific test
ape console                    → interactive Python REPL
ape run scripts/deploy.py      → deploy script

Uses ape.accounts, ape.project. Plugin system: ape-solidity, ape-hardhat, ape-alchemy.
# tests/test_my_token.py
import ape
import pytest

@pytest.fixture
def token(owner):
    return owner.deploy(ape.project.MyToken, 1000)

@pytest.fixture
def owner(accounts):
    return accounts[0]

def test_transfer(token, owner, accounts):
    token.transfer(accounts[1], 50, sender=owner)
    assert token.balanceOf(accounts[1]) == 50

def test_revert_insufficient(token, accounts):
    with ape.reverts("Insufficient balance"):
        token.transfer(accounts[0], 1, sender=accounts[1])
# scripts/deploy.py
from ape import project, accounts

def main():
    deployer = accounts.load("my-account")
    token = deployer.deploy(project.MyToken, 1000)
    print(f"Deployed to: {token.address}")
